   19 -- Definitions of all the ASTs of Llama
   20 
   21 type AST t = [Either (LetDef t) (TypeDef t)]

   23 -- This is useful for repl
   24 data ProgramOrExpr t = Program (AST t)
   25                      | Expression (Expr t)
   26   deriving (Eq, Show)

   28 -- Definitions
   29 data LetDef b = Let [Def b] b
   30               | LetRec [Def b] b
   31   deriving (Eq, Show, Functor, Foldable, Traversable)
   32 
   33 data Def b = FunDef Identifier [Param b] (Expr b) b
   34            | FunDefTyped Identifier [Param b] (Type b) (Expr b) b
   35            | VarDef Identifier b
   36            | VarDefTyped Identifier (Type b) b
   37            | ArrayDef Identifier [Expr b] b
   38            | ArrayDefTyped Identifier [Expr b] (Type b) b
   39   deriving (Eq, Show, Functor, Foldable, Traversable)
   40 
   41 data Param b = Param Identifier b
   42              | TypedParam Identifier (Type b) b
   43   deriving (Eq, Show, Functor, Foldable, Traversable)
   44 
   45 -- Types
   46 data TypeDef b = TypeDef [TDef b] b
   47   deriving (Eq, Show, Functor, Foldable, Traversable)
   48 
   49 data TDef b = TDef Identifier [Constr b] b
   50   deriving (Eq, Show, Functor, Foldable, Traversable)
   51 
   52 data Constr b = Constr ConstrIdentifier [Type b] b
   53   deriving (Eq, Show, Functor, Foldable, Traversable)
   54 
   55 data Type b = Type (TypeF (Type b)) b
   56   deriving (Eq, Show, Functor, Foldable, Traversable)
   57 
   58 data TypeF t = UnitType | IntType | CharType | BoolType | FloatType
   59              | FunType t t
   60              | RefType t
   61              | ArrayType Int t
   62              | UserDefinedType Identifier
   63   deriving (Eq, Ord, Show, Functor, Foldable, Traversable)
   64 
   65 -- Expressions
   66 data Expr b = Expr (ExprF (Expr b)) b
   67             | NewType (Type b) b
   68             | LetIn (LetDef b) (Expr b) b
   69             | MatchExpr (Expr b) [Clause b] b
   70   deriving (Eq, Show, Functor, Foldable, Traversable)
   71 
   72 data ExprF e = IntCExpr IntConstant
   73           | FloatCExpr FloatConstant
   74           | CharCExpr CharConstant
   75           | StringCExpr StringConstant
   76           | TrueCExpr
   77           | FalseCExpr
   78           | UnitCExpr
   79           | ConstExpr Identifier
   80           | ConstConstrExpr ConstrIdentifier
   81           | UnOpExpr UnOp e
   82           | BinOpExpr BinOp e e
   83           | FunAppExpr Identifier [e]
   84           | ConstrAppExpr ConstrIdentifier [e]
   85           | ArrayAccess Identifier [e]
   86           | ArrayDim Identifier Int
   87           | DeleteExpr e
   88           | BeginExpr e
   89           | IfThenExpr e e
   90           | IfThenElseExpr e e e
   91           | WhileExpr e e
   92           | ForExpr Identifier e e e
   93           | ForDownExpr Identifier e e e
   94   deriving (Eq, Show, Functor, Foldable, Traversable)
   95 
   96 data UnOp = PlusUnOp | MinusUnOp
   97           | PlusFloatUnOp | MinusFloatUnOp
   98           | BangOp | NotOp
   99   deriving (Eq, Show)
  100 
  101 data BinOp = PlusOp | MinusOp | TimesOp | DivOp
  102            | PlusFloatOp | MinusFloatOp | TimesFloatOp | DivFloatOp
  103            | ModOp | ExpOp
  104            | EqOp | NotEqOp
  105            | LTOp | GTOp | LEqOp | GEqOp
  106            | NatEqOp | NotNatEqOp
  107            | AndOp | OrOp
  108            | SemicolonOp | AssignMutableOp
  109   deriving (Eq, Show)
  110 
  111 data Clause b = Match (Pattern b) (Expr b) b
  112   deriving (Eq, Show, Functor, Foldable, Traversable)
  113 
  114 data PatternSign = NoSign | Plus | Minus
  115   deriving (Eq, Show)
  116 
  117 data Pattern b = Pattern (PatternF (Pattern b)) b
  118   deriving (Eq, Show, Functor, Foldable, Traversable)
  119 
  120 data PatternF p = IntConstPattern PatternSign IntConstant
  121              | FloatConstPattern PatternSign FloatConstant
  122              | CharConstPattern CharConstant
  123              | TruePattern | FalsePattern
  124              | IdPattern Identifier
  125              | ConstrPattern ConstrIdentifier [p]
  126   deriving (Eq, Show, Functor, Foldable, Traversable)

  128 -- Utils for fmapping all tags
  129 
  130 mapAST :: (a -> b) -> AST a -> AST b
  131 mapAST f = map g where
  132   g (Left l)  = Left (fmap f l)
  133   g (Right t) = Right (fmap f t)

  135 -- Instantiations of the node classes
  136 instance Node LetDef where
  137   tag (Let _ b)    = b
  138   tag (LetRec _ b) = b
  139   mapTag f (Let defs b)    = Let defs (f b)
  140   mapTag f (LetRec defs b) = LetRec defs (f b)
  141 instance Node Def where
  142   tag (FunDef _ _ _ b)        = b
  143   tag (FunDefTyped _ _ _ _ b) = b
  144   tag (VarDef _ b)            = b
  145   tag (VarDefTyped _ _ b)     = b
  146   tag (ArrayDef _ _ b)        = b
  147   tag (ArrayDefTyped _ _ _ b) = b
  148   mapTag f (FunDef i p e b)        = FunDef i p e (f b)
  149   mapTag f (FunDefTyped i p e t b) = FunDefTyped i p e t (f b)
  150   mapTag f (VarDef i b)            = VarDef i (f b)
  151   mapTag f (VarDefTyped i t b)     = VarDefTyped i t (f b)
  152   mapTag f (ArrayDef i e b)        = ArrayDef i e (f b)
  153   mapTag f (ArrayDefTyped i e t b) = ArrayDefTyped i e t (f b)
  154 instance Node Param where
  155   tag (Param _ b)        = b
  156   tag (TypedParam _ _ b) = b
  157   mapTag f (Param i b)        = Param i (f b)
  158   mapTag f (TypedParam i t b) = TypedParam i t (f b)
  159 instance Node TypeDef where
  160   tag (TypeDef _ b) = b
  161   mapTag f (TypeDef tDef b) = TypeDef tDef (f b)
  162 instance Node TDef where
  163   tag (TDef _ _ b) = b
  164   mapTag f (TDef i cs b) = TDef i cs (f b)
  165 instance Node Constr where
  166   tag (Constr _ _ b) = b
  167   mapTag f (Constr i ts b) = Constr i ts (f b)
  168 instance Node Type where
  169   tag (Type _ b) = b
  170   mapTag f (Type tf b) = Type tf (f b)
  171 instance Node Expr where
  172   tag (Expr _ b) = b
  173   tag (NewType _ b) = b
  174   tag (LetIn _ _ b) = b
  175   tag (MatchExpr _ _ b) = b
  176   mapTag f (Expr ef b) = Expr ef (f b)
  177   mapTag f (NewType e b) = NewType e (f b)
  178   mapTag f (LetIn l e b) = LetIn l e (f b)
  179   mapTag f (MatchExpr e cs b) = MatchExpr e cs (f b)
  180 
  181 instance Node Clause where
  182   tag (Match _ _ b) = b
  183   mapTag f (Match p e b) = Match p e (f b)
  184 instance Node Pattern where
  185   tag (Pattern _ b) = b
  186   mapTag f (Pattern pf b) = Pattern pf (f b)

  188 instance NameDef Def where

  190   ide (FunDefTyped i _ _ _ _) = i
  191   ide (VarDef i _)            = i

  199   ide (TDef i _ _) = i
