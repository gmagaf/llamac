   32 class Pretty a where
   33   prettyPrec :: Int -> a -> ShowS
   34   prettyPrec _ a s = pretty a ++ s
   35   pretty :: a -> String
   36   pretty a = prettyPrec 0 a ""
   37   showPretty :: a -> ShowS
   38   showPretty = showString . pretty
   39 
   40 instance Pretty Token where
   41   pretty t = case t of
   42     T_id v           -> v
   43     T_id_constr v    -> v
   44     T_const_int v    -> show v
   45     T_const_float v  -> show v
   46     T_const_char v   -> '\'' : v : "\'"
   47     T_const_string v -> "\"" ++ v ++ "\""
   48     keyword          -> show keyword
   49

   68 prettySepList :: Pretty a => String -> [a] -> String
   69 prettySepList _ []     = ""
   70 prettySepList _ [x]    = pretty x
   71 prettySepList s (x:xs) = showPretty x . showString s $ prettySepList s xs
   72 
   73 prettyPrecSepList :: Pretty a => Int -> String -> [a] -> ShowS
   74 prettyPrecSepList _ _ []     = id
   75 prettyPrecSepList d _ [x]    = prettyPrec d x
   76 prettyPrecSepList d s (x:xs) = prettyPrec d x . showString s . prettyPrecSepList d s xs

  106 instance Pretty t => Pretty (TypeF t) where
  107   prettyPrec d tf =
  108     let
  109       ref_prec = 3
  110       array_prec = 2
  111       fun_prec = 1
  112       showsStars 1 = showPretty T_times
  113       showsStars s = showsStars (s - 1) .
  114         showPretty T_comma . showString " " . showPretty T_times
  115     in case tf of
  116       UnitType -> showPretty T_unit
  117       IntType -> showPretty T_int
  118       CharType -> showPretty T_char
  119       BoolType -> showPretty T_bool
  120       FloatType -> showPretty T_float
  121       UserDefinedType i -> prettyId i
  122       RefType u -> showParen (always || d > ref_prec) $
  123             prettyPrec (ref_prec + 1) u .
  124             showString " " .
  125             showPretty T_ref
  126       ArrayType 1 u -> showParen (always || d > array_prec) $
  127             showPretty T_array . showString " " .
  128             showPretty T_of . showString " " .
  129             prettyPrec (array_prec + 1) u
  130       ArrayType n u -> showParen (always || d > array_prec) $
  131             showPretty T_array . showString " " .
  132             showPretty T_lbracket . showsStars n . showPretty T_rbracket .
  133             showString " " . showPretty T_of . showString " " .
  134             prettyPrec (array_prec + 1) u
  135       FunType u v -> showParen (always || d > fun_prec) $
  136             prettyPrec (fun_prec + 1) u .
  137             showString " " . showPretty T_arrow . showString " " .
  138             prettyPrec fun_prec v

  147 instance Pretty (Def b) where
  148   pretty def = case def of
  149     FunDef i ps e _ -> prettyId i . showString sep . prettyPrecSepList 0 " " ps .
  150       showString " " . showPretty T_equals . showString " " $ pretty e where
  151         sep = if null ps then "" else " "
  152     FunDefTyped i ps t e _ -> prettyId i . showString sep . prettyPrecSepList 0 " " ps .
  153       showString " " . showPretty T_colon . showString " " . showPretty t .
  154       showString " " . showPretty T_equals . showString " " $ pretty e where
  155         sep = if null ps then "" else " "
  156     VarDef i _ -> showPretty T_mutable . showString " " $ prettyId i ""
  157     VarDefTyped i t _ -> showPretty T_mutable . showString " " . prettyId i .
  158       showString " " . showPretty T_colon . showString " " $ showPretty t ""
  159     ArrayDef i es _ -> showPretty T_mutable . showString " " . prettyId i .
  160       showString " " . showPretty T_lbracket . prettyPrecSepList 0 ", " es $
  161       showPretty T_rbracket ""
  162     ArrayDefTyped i es t _ -> showPretty T_mutable . showString " " . prettyId i .
  163       showString " " . showPretty T_lbracket . prettyPrecSepList 0 ", " es .
  164       showPretty T_rbracket . showString " " . showPretty T_colon .
  165       showString " " $ showPretty t ""

  173 instance Pretty (Expr b) where
  174   prettyPrec d expr =
  175     let new_prec = 15
  176         let_prec = 0
  177     in case expr of
  178       Expr e _ -> prettyPrec d e
  179       NewType t _ -> showParen (always || d > new_prec) $
  180         showPretty T_new . showString " " .
  181         showPretty t
  182       LetIn def u _ -> showParen (always || d > let_prec) $
  183         prettyPrec d def . showString " " . showPretty T_in .
  184         showString " " . prettyPrec (let_prec + 1) u
  185       MatchExpr u cs _ -> showPretty T_match . showString " " .
  186         showPretty u . showString " " . showPretty T_with . showString "\n" .
  187         prettyPrecSepList 0 ("\n" ++ pretty T_bar ++ " ") cs . showString "\n" .
  188         showPretty T_end

  190 instance Pretty e => Pretty (ExprF e) where
  191   prettyPrec d e =
  192     let array_access_prec = 14
  193         bang_prec = 13
  194         app_prec = 12
  195         un_op_prec = 11
  196         else_prec = 3
  197         then_prec = 2
  198         if_prec = 2
  199     in case e of
  200       IntCExpr i -> prettyIntC i
  201       FloatCExpr f -> prettyFloatC f
  202       CharCExpr c -> prettyCharC c
  203       StringCExpr s -> prettyStringC s
  204       TrueCExpr -> showPretty T_true
  205       FalseCExpr -> showPretty T_false
  206       UnitCExpr -> showPretty T_lparen . showPretty T_rparen
  207       ConstExpr i -> prettyId i
  208       ConstConstrExpr i -> prettyConstrId i
  209       ArrayDim i 1 -> showParen (always || d > un_op_prec) $
  210         showPretty T_dim . showString " " . prettyId i
  211       ArrayDim i n -> showParen (always || d > un_op_prec) $
  212         showPretty T_dim . showString " " . prettyIntC n .
  213         showString " " . prettyId i
  214       ArrayAccess i es -> showParen (always || d > array_access_prec) $
  215         prettyId i . showPretty T_lbracket .
  216         prettyPrecSepList 0 ", " es .
  217         showPretty T_rbracket
  218       FunAppExpr i ps -> showParen (always || (d > app_prec && not (null ps))) $
  219         prettyId i . showString sep .
  220         prettyPrecSepList (app_prec + 1) " " ps where
  221           sep = if null ps then "" else " "
  222       ConstrAppExpr i ps -> showParen (always || (d > app_prec && not (null ps))) $
  223         prettyConstrId i . showString sep .
  224         prettyPrecSepList (app_prec + 1) " " ps where
  225           sep = if null ps then "" else " "
  226       UnOpExpr BangOp u -> showParen (always || d > bang_prec) $
  227         showPretty T_bang . prettyPrec (bang_prec + 1) u
  228       UnOpExpr NotOp u -> showParen (always || d > un_op_prec) $
  229         showPretty T_not . showString " " . prettyPrec (un_op_prec + 1) u
  230       UnOpExpr op u -> showParen (always || d > un_op_prec) $
  231         showPretty (opToTok op) . prettyPrec (un_op_prec + 1) u
  232       DeleteExpr u -> showParen (always || d > un_op_prec) $
  233         showPretty T_delete . showString " " . prettyPrec (un_op_prec + 1) u
  234       BinOpExpr op u w ->
  235         let (assoc, p, tok) = binOpPrec op
  236         in prettyBinOpExp assoc p tok d u w
  237       IfThenExpr u v -> showParen (always || d > if_prec) $
  238         showPretty T_if . showString " " . prettyPrec (if_prec + 1) u . showString " " .
  239         showPretty T_then . showString " " . prettyPrec then_prec v
  240       IfThenElseExpr u v w -> showParen (always || d > if_prec) $
  241         showPretty T_if . showString " " . prettyPrec (if_prec + 1) u . showString " " .
  242         showPretty T_then . showString " " .
  243         prettyPrec (then_prec + 1) v . showString " " .
  244         showPretty T_else . showString " " . prettyPrec (else_prec + 1) w
  245       BeginExpr u -> showParen always $
  246         showPretty T_begin . showString " " . showPretty u .
  247         showString " " . showPretty T_end
  248       WhileExpr u v -> showParen always $
  249         showPretty T_while . showString " " .
  250         showPretty u . showString " " . showPretty T_do .
  251         showString " " . showPretty v . showString " " . showPretty T_done
  252       ForExpr i u v w -> showParen always $
  253         showPretty T_for . showString " " . prettyId i .
  254         showString " " . showPretty T_equals . showString " " .
  255         showPretty u . showString " " . showPretty T_to . showString " " .
  256         showPretty v . showString " " . showPretty T_do . showString " " .
  257         showPretty w . showString " " . showPretty T_done
  258       ForDownExpr i u v w -> showParen always $
  259         showPretty T_for . showString " " . prettyId i .
  260         showString " " . showPretty T_equals . showString " " .
  261         showPretty u . showString " " . showPretty T_downto . showString " " .
  262         showPretty v . showString " " . showPretty T_do . showString " " .
  263         showPretty w . showString " " . showPretty T_done

  265 data Assoc = L | R | Non
  266   deriving Eq

  268 prettyBinOpExp :: Pretty e => Assoc -> Int -> Token -> Int -> e -> e -> ShowS
  269 prettyBinOpExp L = prettyBinOpExpL
  270 prettyBinOpExp R = prettyBinOpExpR
  271 prettyBinOpExp Non = prettyBinOpExpNon
  272 
  273 prettyBinOpExpL :: Pretty e => Int -> Token -> Int -> e -> e -> ShowS
  274 prettyBinOpExpL prec tok d u w = showParen (always || d > prec) $
  275   prettyPrec prec u . showString " " .
  276   showPretty tok . showString " " .
  277   prettyPrec (prec + 1) w
  278 
  279 prettyBinOpExpR :: Pretty e => Int -> Token -> Int -> e -> e -> ShowS
  280 prettyBinOpExpR prec tok d u w = showParen (always || d > prec) $
  281   prettyPrec (prec + 1) u . showString " " .
  282   showPretty tok . showString " " .
  283   prettyPrec prec w
  284 
  285 prettyBinOpExpNon :: Pretty e => Int -> Token -> Int -> e -> e -> ShowS
  286 prettyBinOpExpNon prec tok d u w = showParen (always || d > prec) $
  287   prettyPrec (prec + 1) u . showString " " .
  288   showPretty tok . showString " " .
  289   prettyPrec (prec + 1) w
