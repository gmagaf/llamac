   43 -- This module contains the defintion of the Symbol table for the compiler
   44 
   45 type Scope = M.Map
   46 
   47 newtype Context k e = Context [Scope k e]
   48     deriving (Show, Functor, Foldable, Traversable)
   49 
   50 -- Symbol table operations
   51 emptyContext :: Context k e
   52 emptyContext = Context []
   53 
   54 partialQuery :: Ord k => Getter e e' -> k -> Context k e -> Maybe e'
   55 partialQuery _ _ (Context []) = Nothing
   56 partialQuery l k (Context (scope:scopes)) =
   57     case M.lookup k scope of
   58         Nothing -> partialQuery l k (Context scopes)
   59         Just e  -> Just (view l e)
   60 
   61 query :: Ord k => k -> Context k e -> Maybe e
   62 query = partialQuery (to id)
   63 
   64 partialInsert :: Ord k => (e' -> e) -> k -> e' -> Context k e -> Context k e
   65 partialInsert mk k e (Context scopes) = case scopes of
   66     []   -> Context [M.insert k (mk e) M.empty]
   67     s:tl -> Context $ updated:tl where
   68         updated = M.insert k (mk e) s
   69 
   70 insert :: Ord k => k -> e -> Context k e -> Context k e
   71 insert = partialInsert id
   72 
   73 partialUpdate :: Ord k => (e' -> e -> e) -> k -> e' -> Context k e -> Context k e
   74 partialUpdate f k e' (Context s) = Context (aux s) where
   75     aux [] = []
   76     aux (scope:scopes) = if M.member k scope
   77         then M.update (Just . f e') k scope : scopes
   78         else scope : aux scopes
   79 
   80 update :: Ord k => k -> e -> Context k e -> Context k e
   81 update = partialUpdate const
   82 
   83 openScope :: Context k e -> Context k e
   84 openScope (Context scopes) = Context $ M.empty:scopes
   85 
   86 closeScope :: Context k e -> Context k e
   87 closeScope (Context scopes) = Context s where
   88     s = if null scopes then [] else tail scopes

   90 -- Definitions for symbol table and table entries
   91 data FullTableEntry e g = FullTableEntry {
   92     _basicInfo :: e,
   93     _optInfo :: Maybe g
   94     } deriving Show
   95 
   96 mkFullTableEntry :: e -> Maybe g -> FullTableEntry e g
   97 mkFullTableEntry = FullTableEntry
   98 
   99 mkBasicEntry :: e -> FullTableEntry e g
  100 mkBasicEntry = flip FullTableEntry Nothing
  101 
  102 basicInfo :: Lens' (FullTableEntry e g) e
  103 basicInfo = lens _basicInfo (\e b -> e{_basicInfo = b})
  104 
  105 optInfo :: Lens' (FullTableEntry e g) (Maybe g)
  106 optInfo = lens _optInfo (\e g -> e{_optInfo = g})
  107 
  108 generated :: Prism' (FullTableEntry e g) (e, g)
  109 generated = prism' (\(e, g) -> FullTableEntry e (Just g)) (\(FullTableEntry e g) -> fmap (e,) g)

  111 data TableEntry
  112     = MutableEntry SymbolType                     -- Type of the mutable variable
  113     | ArrayEntry SymbolType Int                   -- Type of the entries, num of dimensions
  114     | FunEntry TypeScheme [Identifier]            -- TypeScheme of the function, params
  115     | ParamEntry SymbolType Identifier            -- Type of the param, function of the param
  116     | PatternEntry SymbolType                     -- Type of the pattern
  117     | ConstrEntry ConstType [ConstType] ConstType -- Type of constructor, params, output type
  118         deriving Show
  119 
  120 newtype TypeTableEntry
  121     = TypeEntry [(ConstrIdentifier, [ConstType])] -- Constructors and arguements
  122         deriving Show
  123 
  124 type NameSpace = Context String (FullTableEntry TableEntry L.Operand)
  125 type TypeSpace = Context String (FullTableEntry TypeTableEntry L.Type)
  126 data SymbolTable = SymbolTable {
  127     _names :: NameSpace,
  128     _types :: TypeSpace
  129     }
  130     deriving Show
  131 makeLenses ''SymbolTable
  132 
  133 emptySymbolTable :: SymbolTable
  134 emptySymbolTable = SymbolTable emptyContext emptyContext

  136 -- Pretty printing of symbol table
  137 instance Pretty TableEntry where
  138     pretty entry = case entry of
  139         MutableEntry t ->
  140             "Mutable var of type: " ++ pretty t
  141         ArrayEntry t dim ->
  142             show dim ++ "-dimensional array of type: " ++ pretty t
  143         FunEntry funType [] ->
  144             "Const of type: " ++ pretty funType
  145         FunEntry funType params ->
  146             "Fun of type: " ++ pretty funType ++ " with params: " ++ intercalate ", " params
  147         ParamEntry t f ->
  148             "Param of type: " ++ pretty t ++ " of function " ++ f
  149         ConstrEntry constrType [] outputType ->
  150             "Constr of " ++ pretty outputType ++
  151             " with type: " ++ pretty constrType
  152         PatternEntry t ->
  153             "Pattern of type " ++ pretty t
  154         ConstrEntry constrType ts outputType ->
  155             "Constr of " ++ pretty outputType ++
  156             " with type: " ++ pretty constrType ++
  157             " with params: (" ++ intercalate ", " (map pretty ts) ++ ")"
  158 
  159 instance (Pretty e, Show g) => Pretty (FullTableEntry e g) where
  160     pretty entry = pretty (view basicInfo entry) ++
  161                    maybe "" (\(_, i) -> " additional info: " ++ show i) (preview generated entry)
  162 
  163 instance Pretty TypeTableEntry where
  164     pretty entry = case entry of
  165         TypeEntry constrs ->
  166             "Type with constrs: " ++ cs where
  167                 f (c, []) = c
  168                 f (c, ps) = c ++ " of " ++ unwords (map pretty ps)
  169                 cs = intercalate ", " (map f constrs)
  170 
  171 instance (Show k, Pretty e) => Pretty (Context k e) where
  172     pretty (Context scopes) =
  173         let -- Utils for each record
  174             toString = bimap show pretty
  175             toLengths = bimap length length
  176             toPaddings (accK, accE) = bimap (max accK) (max accE)
  177             addPadding v l = v ++ replicate (max 0 (l - length v)) ' '
  178             -- Utils for each scope
  179             scopeToString = M.foldMapWithKey (\k e -> [toString (k, e)])
  180             stringsToLengths = map toLengths
  181             lengthsToPaddings = foldr toPaddings (0, 0)
  182             -- Compute values for all context
  183             strings = map scopeToString scopes
  184             lengths = map stringsToLengths strings
  185             paddings = map lengthsToPaddings lengths
  186             -- Compute total lengths
  187             (lk', le') = lengthsToPaddings paddings
  188             (lk, le) = (max lk' $ length "Keys", max le' $ length "Entries")
  189             totalLength = 1 + 1 + lk + 1 + 1 + 1 + le + 1 + 1
  190             line = replicate totalLength '-' ++ "\n"
  191             -- Utils for printing a line
  192             printLine (k, e) acc = "| " ++ addPadding k lk ++ " | " ++ addPadding e le ++ " |\n" ++ acc
  193             printScope scope acc = foldr printLine (line ++ acc) scope
  194             scopesTables = foldr printScope "" strings
  195         in line ++ printLine ("Keys", "Entries")  (line ++ scopesTables)
  196 
  197 instance Pretty SymbolTable where
  198     pretty st =
  199         "Types Namespace\n" ++
  200         pretty (view types st) ++
  201         "Names Namespace\n" ++
  202         pretty (view names st)
