   31 -- Symbol Types are types including type variables
   32 data SymbolType = SymType (TypeF SymbolType)
   33                 | TVar Int
   34     deriving (Show, Eq)
   35 
   36 instance Pretty SymbolType where
   37     prettyPrec d (SymType t) = prettyPrec d t
   38     prettyPrec _ (TVar i)    = showString $ "@" ++ show i
   39 
   40 -- Some convertion utils
   41 typeTo :: (TypeF a -> a) -> Type b -> a
   42 typeTo alg (Type tf _) = alg (fmap (typeTo alg) tf)

   44 constTypeToSymbolType :: ConstType -> SymbolType
   45 constTypeToSymbolType (ConstType tf) = SymType $ fmap constTypeToSymbolType tf

   47 -- Type schemes are polymorphic types
   48 data TypeScheme = MonoType SymbolType
   49                 | AbsType Int TypeScheme
   50     deriving (Show, Eq)
   51 
   52 instance Pretty TypeScheme where
   53     prettyPrec d (MonoType t)  = prettyPrec d t
   54     prettyPrec d (AbsType v t) =
   55         showString ("forall @" ++ show v ++ ". ") .
   56         prettyPrec d t

   58 -- Recursion Utils
   59 bottomUp :: (SymbolType -> SymbolType) -> SymbolType -> SymbolType
   60 bottomUp f (SymType t) = f (SymType $ fmap (bottomUp f) t)
   61 bottomUp f v@(TVar _)  = f v

   67 cataUn :: (t -> TypeF t) -> (TypeF a -> a) -> t -> a
   68 cataUn unT alg tf = alg (fmap (cataUn unT alg) (unT tf))

   70 cataUnM :: Monad m => (t -> TypeF t) -> (TypeF a -> m a) -> t -> m a
   71 cataUnM unT alg tf = mapM (cataUnM unT alg) (unT tf) >>= alg

   73 cata :: (TypeF a -> a, Int -> a) -> SymbolType -> a
   74 cata alg@(f, _) (SymType t) = f $ fmap (cata alg) t
   75 cata (_, g) (TVar v) = g v
   76 
   77 cataM :: Monad m => (TypeF a -> m a, Int -> m a) -> SymbolType -> m a
   78 cataM alg@(f, _) (SymType t) = mapM (cataM alg) t >>= f
   79 cataM (_, g) (TVar v) = g v

   81 -- Type theoretic utils
   82 subst :: Int -> SymbolType -> SymbolType -> SymbolType
   83 subst v t = bottomUp f where
   84     f :: SymbolType -> SymbolType
   85     f s'@(TVar u) | v == u = t
   86                   | otherwise = s'
   87     f s' = s'
   88 
   89 substScheme :: Int -> SymbolType -> TypeScheme -> TypeScheme
   90 substScheme v r (MonoType t) = MonoType $ subst v r t
   91 substScheme v r (AbsType u t) | v == u    = AbsType u t
   92                               | otherwise = AbsType u $ substScheme v r t

   94 tvarsInType :: SymbolType -> [Int]
   95 tvarsInType st = cata (aux, (:)) st [] where
   96     aux :: TypeF ([Int] -> [Int]) -> [Int] -> [Int]
   97     aux (FunType f g)   = f . g
   98     aux (ArrayType _ f) = f
   99     aux (RefType f)     = f
  100     aux _               = id
  101 
  102 tvarsInScheme :: TypeScheme -> [Int]
  103 tvarsInScheme (MonoType t)  = tvarsInType t
  104 tvarsInScheme (AbsType _ t) = tvarsInScheme t
  105 
  106 freeVarsInScheme :: TypeScheme -> [Int]
  107 freeVarsInScheme t = S.toList $ S.difference allvars bound where
  108     allvars = S.fromList $ tvarsInScheme t
  109     bound = S.fromList (aux t)
  110     aux (MonoType _)  = []
  111     aux (AbsType v s) = v : aux s

  113 tvarInType :: Int -> SymbolType -> Bool
  114 tvarInType v (TVar u)
  115                 | v == u = True
  116                 | otherwise = False
  117 tvarInType v (SymType t) = any (tvarInType v) t
  118 
  119 notVarInType :: Int -> SymbolType -> Bool
  120 notVarInType v = not . tvarInType v

  122 -- Fun Types Utils
  123 paramsToFun :: (TypeF a -> a) -> [a] -> a -> a
  124 paramsToFun _ [] out       = out
  125 paramsToFun alg (t:ts) out = alg (FunType t (paramsToFun alg ts out))
  126 
  127 funToTypes :: (a -> Either (TypeF a) a) -> a -> [a]
  128 funToTypes coalg = reverse . aux [] where
  129     aux acc t = case coalg t of
  130         Left (FunType t1 t2) -> aux (t1:acc) t2
  131         _                    -> t:acc
  132 
  133 funToArgs :: (a -> Either (TypeF a) a) -> a -> [a]
  134 funToArgs coalg s =
  135     let ts = funToTypes coalg s
  136         f [] = []
  137         f [_] = []
  138         f (x:xs) = x : f xs
  139     in if null ts then [] else f ts
  140 
  141 outFunType :: (a -> Either (TypeF a) a) -> a -> a
  142 outFunType out t = case out t of
  143     Left (FunType _ t2) -> outFunType out t2
  144     _ -> t

  150 stCoAlg (SymType t) = Left t
