   96 instance Show Token where
   97   show t = case t of
   98     -- Keywords
   99     T_and               -> "and"
  100     T_array             -> "array"
  101     T_begin             -> "begin"
  102     T_bool              -> "bool"
  103     T_char              -> "char"
  104     T_delete            -> "delete"
  105     T_dim               -> "dim"
  106     T_do                -> "do"
  107     T_done              -> "done"
  108     T_downto            -> "downto"
  109     T_else              -> "else"
  110     T_end               -> "end"
  111     T_false             -> "false"
  112     T_float             -> "float"
  113     T_for               -> "for"
  114     T_if                -> "if"
  115     T_in                -> "in"
  116     T_int               -> "int"
  117     T_let               -> "let"
  118     T_match             -> "match"
  119     T_mod               -> "mod"
  120     T_mutable           -> "mutable"
  121     T_new               -> "new"
  122     T_not               -> "not"
  123     T_of                -> "of"
  124     T_rec               -> "rec"
  125     T_ref               -> "ref"
  126     T_then              -> "then"
  127     T_to                -> "to"
  128     T_true              -> "true"
  129     T_type              -> "type"
  130     T_unit              -> "unit"
  131     T_while             -> "while"
  132     T_with              -> "with"
  133     -- Identifiers
  134     T_id v              -> "id:" ++ v
  135     T_id_constr v       -> "constrId:" ++ v
  136     -- Constants
  137     T_const_int v       -> "intC:" ++ show v
  138     T_const_float v     -> "floatC:" ++ show v
  139     T_const_char v      -> "charC:" ++ [v]
  140     T_const_string v    -> "stringC:" ++ v
  141     -- Symbols
  142     T_arrow             -> "->"
  143     T_equals            -> "="
  144     T_bar               -> "|"
  145     T_plus              -> "+"
  146     T_minus             -> "-"
  147     T_times             -> "*"
  148     T_div               -> "/"
  149     T_plus_float        -> "+."
  150     T_minus_float       -> "-."
  151     T_times_float       -> "*."
  152     T_div_float         -> "/."
  153     T_exp               -> "**"
  154     T_bang              -> "!"
  155     T_semicolon         -> ";"
  156     T_and_op            -> "&&"
  157     T_or_op             -> "||"
  158     T_not_equals        -> "<>"
  159     T_less_than         -> "<"
  160     T_greater_than      -> ">"
  161     T_less_than_eq      -> "<="
  162     T_greater_than_eq   -> ">="
  163     T_nat_eq_op         -> "=="
  164     T_not_nat_eq_op     -> "!="
  165     T_assign_mutable    -> ":="
  166     T_lparen            -> "("
  167     T_rparen            -> ")"
  168     T_lbracket          -> "["
  169     T_rbracket          -> "]"
  170     T_comma             -> ","
  171     T_colon             -> ":"
  172     T_eof               -> "eof"
