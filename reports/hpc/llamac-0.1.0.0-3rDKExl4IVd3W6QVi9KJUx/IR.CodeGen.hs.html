   22 genAST :: AST SemanticTag -> Parser ()
   23 genAST [] = return ()
   24 genAST (Left def:ast) = genDefs def >> genAST ast
   25 genAST (Right _:ast) = genAST ast -- TODO: Define types gen

   27 genDefs :: LetDef SemanticTag -> Parser ()
   28 genDefs (Let defs _) = do
   29     gens <- mapM genDef defs
   30     registerDefs defs gens
   31 genDefs (LetRec defs _) = mdo
   32     registerDefs defs gens
   33     gens <- mapM genDef defs
   34     return ()
   35 
   36 registerDefs :: [Def SemanticTag] -> [L.Operand] -> Parser ()
   37 registerDefs [] _ = return ()
   38 registerDefs (f@(FunDef fname params _ _):ds) ops = do -- TODO: Define register for the rest of the cases
   39     s <- getDefScheme f
   40     symbols . names %= insert fname (mkFullTableEntry (FunEntry s (map ide params)) (Just (head ops)))
   41     registerDefs ds (tail ops)

   43 genParam :: Param SemanticTag -> Parser (L.Type, L.ParameterName)
   44 genParam p = do
   45     t <- getNodeType p
   46     lt <- genType t
   47     return (lt, fromString (ide p))

   49 genDef :: Def SemanticTag -> Parser L.Operand
   50 genDef n@(FunDef fname params body _) = do -- TODO: Define defs gen
   51     scheme <- getDefScheme n
   52     st <- inst scheme
   53     let retsty = getRetStType st
   54     retty <- genType retsty
   55     genParams <- mapM genParam params
   56     function (L.mkName fname) genParams retty genBody where
   57         registerFormalParam :: (Param SemanticTag, L.Operand) -> Parser ()
   58         registerFormalParam (p, op) = do
   59             t <- getNodeType p
   60             symbols . names %= insert (ide p) (mkFullTableEntry (ParamEntry t (ide p)) (Just op))
   61         genBody :: [L.Operand] -> Parser ()
   62         genBody ops = do
   63             openScopeInNames
   64             let ps = zip params ops
   65             mapM_ registerFormalParam ps
   66             _entry <- L.block `named` "entry"
   67             res <- genExpr body
   68             L.ret res
   69             closeScopeInNames

   71 genExpr :: Expr SemanticTag -> Parser L.Operand -- TODO: Define expr gen
   72 genExpr (Expr ef _) = case ef of
   73     TrueCExpr    -> return . boolConst $ True
   74     FalseCExpr   -> return . boolConst $ False
   75     CharCExpr c  -> return . charConst $ c
   76     IntCExpr i   -> return . intConst $ i
   77     FloatCExpr f -> return . floatConst $ f
   78     BinOpExpr op e1 e2 -> genBinOp op e1 e2
   79     ConstExpr x -> do
   80                 ns <- use (symbols . names)
   81                 case query x ns of
   82                     Just fte -> case view optInfo fte of
   83                         Just o -> return o
   84                         Nothing -> throwCGenError $ "Const: " ++ x ++ " is not yet generated"
   85                     Nothing -> throwCGenError $ "Const: " ++ x ++ " is not yet in symbol table"
   86 
   87 genBinOp :: BinOp -> Expr SemanticTag -> Expr SemanticTag -> Parser L.Operand
   88 genBinOp op lhs rhs = case op of
   89     PlusOp -> do
   90         l <- genExpr lhs
   91         r <- genExpr rhs
   92         L.add l r
   93     MinusOp -> do
   94         l <- genExpr lhs
   95         r <- genExpr rhs
   96         L.sub l r
