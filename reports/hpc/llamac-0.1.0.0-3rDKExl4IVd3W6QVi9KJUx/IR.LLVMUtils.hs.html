   38 -- Utility for easily building global functions
   39 functionDefaults :: L.Linkage -> Name -> [(Type, Name)] -> Type -> [BasicBlock] -> Global
   40 functionDefaults l n ps ret bbs =
   41   Function {
   42     linkage = l,
   43     visibility = V.Default,
   44     dllStorageClass = Nothing,
   45     callingConvention = CC.C,
   46     returnAttributes = [],
   47     returnType = ret,
   48     name = n,
   49     parameters = ([Parameter ty pn [] | (ty, pn) <- ps], False),
   50     functionAttributes = [],
   51     section = Nothing,
   52     comdat = Nothing,
   53     alignment = 0,
   54     garbageCollectorName = Nothing,
   55     prefix = Nothing,
   56     basicBlocks = bbs,
   57     personalityFunction = Nothing,
   58     metadata = []
   59   }

   61 -- Utility for declaring external functions
   62 extern
   63   :: IR.MonadModuleBuilder m
   64   => String           -- ^ Definition identifier
   65   -> [(Type, String)] -- ^ Parameter types and identifiers pairs
   66   -> Type             -- ^ Type
   67   -> m Operand
   68 extern nm argtys retty = do
   69   let funty = ptr $ FunctionType retty (map fst argtys) False
   70   IR.emitDefn $ GlobalDefinition (functionDefaults L.External (mkName nm) (map (B.second mkName) argtys) retty [])
   71   pure $ ConstantOperand $ C.GlobalReference funty (mkName nm)

   73 -- Util to run a parser within a parser
   74 -- with a given initial IRBuilderState
   75 -- This is used to generate the basic blocks
   76 -- from a parser
   77 runIRBuilder :: Monad m => IR.IRBuilderState -> ParserT e ParserState m a -> ParserT e ParserState m (a, [BasicBlock])
   78 runIRBuilder irs p = parserT $ \ps -> do
   79   ~(eitherNames, s') <- runParserT (p <* IR.block) (set (cgen_state . irState) irs ps)
   80   case eitherNames of
   81     Left e  -> return (Left e, s')
   82     Right a -> do
   83       let blocks = getSnocList . IR.builderBlocks . view (cgen_state . irState) $ s'
   84       return (Right (a, blocks), s')

   86 -- Utility for emitting a function
   87 function
   88   :: Name                       -- ^ Function name
   89   -> [(Type, IR.ParameterName)] -- ^ Parameter types and name suggestions
   90   -> Type                       -- ^ Return type
   91   -> ([Operand] -> Parser ())   -- ^ Function body builder
   92   -> Parser Operand
   93 function label argtys retty body = do
   94   flabel <- case label of
   95     Name l   -> IR.fresh `IR.named` l
   96     UnName _ -> IR.fresh
   97   irs <- use (cgen_state . irState)
   98   let irs' = IR.emptyIRBuilder{
   99                   IR.builderUsedNames = IR.builderUsedNames irs,
  100                   IR.builderNameSuggestion = IR.builderNameSuggestion irs}
  101   ~(tyNamePairs, blocks) <- runIRBuilder irs' $ do
  102       (formalVars, tyNamePairs) <- genParams argtys
  103       body formalVars
  104       return tyNamePairs
  105   let def = GlobalDefinition (functionDefaults L.Internal flabel tyNamePairs retty blocks)
  106       funty = ptr $ FunctionType retty (fst <$> argtys) False
  107   IR.emitDefn def
  108   pure $ ConstantOperand $ C.GlobalReference funty flabel where
  109     genParams [] = return ([], [])
  110     genParams ((ty, n):ps) = do
  111       param <- case n of
  112           IR.NoParameterName -> IR.fresh
  113           IR.ParameterName p -> IR.fresh `IR.named` p
  114       (\(x, y) (xs, ys) -> (x:xs, y:ys)) (LocalReference ty param, (ty, param)) <$> genParams ps
