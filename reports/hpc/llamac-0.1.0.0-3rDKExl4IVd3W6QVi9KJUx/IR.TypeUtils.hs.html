   18 charStar :: L.Type
   19 charStar = L.ptr L.i8
   20 
   21 -- Util for computing the size needed
   22 -- to hold data of a type
   23 constTypeSize :: ConstType -> Word64
   24 constTypeSize = cataUn unConstType sizeOfType
   25 
   26 sizeOfType :: TypeF Word64 -> Word64
   27 sizeOfType tf = case tf of
   28     UnitType          -> 0
   29     BoolType          -> 1
   30     CharType          -> 1
   31     IntType           -> 4
   32     FloatType         -> 8
   33     FunType _ _       -> 8 -- the rest will be sized as pointers
   34     RefType _         -> 8 -- to the corresponding data in memory
   35     ArrayType _ _     -> 8
   36     UserDefinedType _ -> 8
   37 
   38 -- Util for generating a constant type
   39 genConstType :: ConstType -> Parser L.Type
   40 genConstType = cataUnM unConstType genTypeF
   41 
   42 genType :: SymbolType -> Parser L.Type
   43 genType = cataM (genTypeF, \_ -> return charStar)
   44 
   45 genTypeF :: TypeF L.Type -> Parser L.Type
   46 genTypeF tf = case tf of
   47     UnitType                                -> return L.void
   48     BoolType                                -> return L.i1
   49     CharType                                -> return L.i8
   50     IntType                                 -> return L.i32
   51     FloatType                               -> return L.float
   52     FunType t1 (L.FunctionType rt2 argt2 _) -> return (L.FunctionType rt2 (t1:argt2) False)
   53     FunType t1 t2                           -> return (L.FunctionType t2 [t1] False)
   54     RefType L.VoidType                      -> return charStar -- (void *) is illegal in llvm
   55     RefType t                               -> return (L.ptr t)
   56     ArrayType _ t                           -> return (L.ptr t)  -- TODO: Define this
   57     UserDefinedType i                       -> do
   58         res <- queryTypeP i
   59         case res of
   60             Just (TypeEntry entry) -> do
   61                 -- res = map (\(ci, ct) -> (ci, map genConstType ct)) tentry
   62                 let sizeOfConstr :: (ConstrIdentifier, [ConstType]) -> Word64
   63                     sizeOfConstr = foldr (\t -> (constTypeSize t +)) 0 . snd
   64                     size = foldr (\c acc -> max acc (sizeOfConstr c)) 0 entry
   65                 return $ L.StructureType True (L.i8:([L.ArrayType size L.i8 | size /= 0])) -- TODO: Define this
   66             Nothing -> throwCGenError ("Unable to find type: " ++ i ++ " to generate")
   67
