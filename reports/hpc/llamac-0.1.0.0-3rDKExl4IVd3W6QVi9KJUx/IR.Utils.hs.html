   30 codegenProgram :: String -> Parser T.Text
   31 codegenProgram mname = do
   32     ms <- use (cgen_state . moduleState)
   33     let defs = L.getSnocList . L.builderDefs $ ms
   34     let mdl = mkModule defs
   35     return (ppllvm mdl) where
   36         mkModule ds = L.defaultModule { L.moduleName = fromString mname, L.moduleDefinitions = ds }
