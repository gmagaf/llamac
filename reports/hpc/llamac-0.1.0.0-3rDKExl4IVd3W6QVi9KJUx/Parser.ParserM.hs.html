   25 -- The compiler's errors
   26 data Error = Error {msg :: String}
   27            | InternalError {msg :: String}
   28            | LexicalError {msg :: String}
   29            | ParsingError {msg :: String}
   30            | SemanticError {msg :: String}
   31            | CodeGenError {msg :: String}
   32     deriving Eq
   33 
   34 instance Show Error where
   35   show (Error s)          = s
   36   show (InternalError s)  = "Internal Compiler Error: " ++ s ++
   37     ". If you see this error please contact the maintainers"
   38   show (LexicalError s)   = "Lexical Error: " ++ s
   39   show (ParsingError s)   = "Parser Error: " ++ s
   40   show (SemanticError s)  = "Semantic Error: " ++ s
   41   show (CodeGenError s)   = "Code Gen Error: " ++ s
   42

   44 -- The monad definition

   75 -- Utils for running a Parser

  104 stackTrace :: String -> Parser a -> Parser a
  105 stackTrace s = catch (throw . \e -> e{msg = msg e ++ "\n\t\t" ++ s})

  107 -- Utils to facilitate the communication with the lexer
  108 liftAlex :: Alex a -> Parser a
  109 liftAlex (Alex f) = do
  110   aState <- getAlexState
  111   case f aState of
  112     Right (aState', a) -> do
  113       putAlexState aState'
  114       return a
  115     Left lexErr        -> throwLexicalError lexErr
  116 
  117 lexerWrap :: (Token -> Parser a) -> Parser a
  118 lexerWrap cont = do
  119   a <- liftAlex alexMonadScan
  120   cont a
