   10 import Common.SymbolTable (SymbolTable, emptySymbolTable)
   11 import Lexer.Lexer (AlexState(..), alexStartPos, alexInitUserState)

   13 import IR.CodeGenState (CodeGenState, initCodeGenState)

   16 -- The state of the parser
   17 data ParserState = ParserState
   18   { _alex_state   :: AlexState     -- lexer's state
   19   , _sem_state    :: SemanticState -- semantic analysis state
   20   , _symbols      :: SymbolTable   -- compiler's symbol table
   21   , _cgen_state   :: CodeGenState  -- code generator's state
   22   }
   23 makeLenses ''ParserState
   24 
   25 instance Show ParserState where
   26   show s = "ParserState {" ++
   27     "alex_state = _"
   28     ++ ", " ++ "sem_state = " ++ show (view sem_state s)
   29     ++ ", " ++ "cgen_state = " ++ show (view cgen_state s)
   30     ++ ", " ++ "symbols = " ++ show (view symbols s)
   31     ++ "}"
   32 
   33 initAlexState :: String -> AlexState
   34 initAlexState input = AlexState {alex_pos = alexStartPos,
   35                         alex_inp = input,
   36                         alex_chr = '\n',
   37                         alex_bytes = [],
   38                         alex_ust = alexInitUserState,
   39                         alex_scd = 0}
   40 
   41 initParserState :: String -> ParserState
   42 initParserState input =
   43     ParserState
   44        { _alex_state = initAlexState input
   45        , _sem_state  = initSemanticState
   46        , _symbols    = emptySymbolTable
   47        , _cgen_state = initCodeGenState
   48        }
