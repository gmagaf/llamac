   26 -- This module defines the Parser monad transformer
   27 
   28 -- The monad transformer definition
   29 newtype ParserT e s m a = ParserT { getParserT :: Except.ExceptT e (State.StateT s m) a }
   30 
   31 -- Constructors
   32 parserT :: (s -> m (Either e a, s)) -> ParserT e s m a
   33 parserT = ParserT . Except.ExceptT . State.StateT
   34 
   35 pureParserT :: Monad m => (s -> (Either e a, s)) -> ParserT e s m a
   36 pureParserT f = ParserT . Except.ExceptT . State.StateT $ (return . f)
   37 
   38 -- State related
   39 evalParserT :: Monad m => ParserT e s m a -> s -> m (Either e a)
   40 evalParserT p = State.evalStateT (Except.runExceptT (getParserT p))
   41 
   42 runParserT :: ParserT e s m a -> s -> m (Either e a, s)
   43 runParserT ~(ParserT (Except.ExceptT (State.StateT f))) = f
   44 
   45 -- Exception related
   46 throw :: Monad m => e -> ParserT e s m a
   47 throw = ParserT . Except.throwE
   48 
   49 withExcept :: Monad m => (e -> e') -> ParserT e s m a -> ParserT e' s m a
   50 withExcept f = ParserT . Except.withExceptT f . getParserT
   51 
   52 catch :: Monad m => (e -> ParserT e' s m a) -> ParserT e s m a -> ParserT e' s m a
   53 catch handle p = ParserT $ Except.catchE (getParserT p) (getParserT . handle)
   54 
   55 instance Functor f => Functor (ParserT e s f) where
   56   fmap :: forall a b. (a -> b) -> ParserT e s f a -> ParserT e s f b
   57   fmap f p = parserT (fmap g . runParserT p) where
   58     g :: (Either e a, s) -> (Either e b, s)
   59     g ~(a, s) = (fmap f a, s)
   60 
   61 instance (Functor m, Monad m) => Applicative (ParserT e s m) where
   62   pure :: a -> ParserT e s m a
   63   pure a = parserT $ \s -> return (Right a, s)
   64   (<*>) :: forall a b. ParserT e s m (a -> b) -> ParserT e s m a -> ParserT e s m b
   65   pf <*> pa = parserT $ \s -> do
   66     ~(ef, s') <- runParserT pf s
   67     case ef of
   68       Left e  -> return (Left e, s')
   69       Right f -> do
   70         ~(ea, s'') <- runParserT pa s'
   71         case ea of
   72           Left e  -> return (Left e, s'')
   73           Right a -> return (Right (f a), s'')
   74 
   75 instance Monad m => Monad (ParserT e s m) where
   76   return :: a -> ParserT e s m a
   77   return a = parserT $ \s -> return (Right a, s)
   78   (>>=) :: ParserT e s m a -> (a -> ParserT e s m b) -> ParserT e s m b
   79   p >>= f = parserT $ \s -> do
   80               ~(ea, s') <- runParserT p s
   81               case ea of
   82                 Left e  -> return (Left e, s')
   83                 Right a -> runParserT (f a) s'
   84 
   85 instance (MonadFix m, Show e) => MonadFix (ParserT e s m) where
   86   mfix :: forall a. (a -> ParserT e s m a) -> ParserT e s m a
   87   mfix f = parserT mf where
   88     mf :: s -> m (Either e a, s)
   89     mf s = mfix (\ ~(ea, _) -> runParserT (f (either bomb id ea)) s)
   90     bomb = error . ("mfix (ParserT): recursive computation returned Left value: " ++) . show
   91 
   92 instance MonadIO m => MonadIO (ParserT e s m) where
   93   liftIO :: IO a -> ParserT e s m a
   94   liftIO io = ParserT (liftIO io)
   95 
   96 instance Monad m => MonadState s (ParserT e s m) where
   97   get :: ParserT e s m s
   98   get = ParserT $ lift State.get
   99   put :: s -> ParserT e s m ()
  100   put s = ParserT $ lift (State.put s)
  101 
  102 instance MonadTrans (ParserT e s) where
  103   lift :: Monad m => m a -> ParserT e s m a
  104   lift ma = parserT f where
  105     f s = do
  106       a <- ma
  107       return (Right a, s)
  108 
  109 instance (Monad m) => MonadIRBuilder (ParserT e ParserState m) where
  110   liftIRState :: S.State IRBuilderState a -> ParserT e ParserState m a
  111   liftIRState s = do
  112     irs <- use (cgen_state . irState)
  113     let (a, irs') = S.runState s irs
  114     cgen_state . irState .= irs'
  115     return a
  116 
  117 instance Monad m => MonadModuleBuilder (ParserT e ParserState m) where
  118   liftModuleState :: State.State ModuleBuilderState a -> ParserT e ParserState m a
  119   liftModuleState s = do
  120     ms <- use (cgen_state . moduleState)
  121     let ~(a, ms') = State.runState s ms
  122     cgen_state . moduleState .= ms'
  123     return a
