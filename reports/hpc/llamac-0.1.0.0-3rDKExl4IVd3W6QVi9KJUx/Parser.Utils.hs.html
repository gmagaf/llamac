   25 -- Various useful parsers

   35 -- Util that initilizes a parser state and runs a parser monad
   36 parseString :: Parser a -> String -> (Either Error a, ParserState)
   37 parseString m s = runParser initState m where
   38   initState :: ParserState
   39   initState = initParserState s

   52 -- Util function for debugging end to end
   53 debug :: String -> IO ()
   54 debug s = do
   55   let (res, state) = parseString (initParseAnalyzeGenM "debug from ghci") s
   56   putStrLn "Semantic State"
   57   print (view sem_state state)
   58   putStrLn "Code Gen State"
   59   print (view cgen_state state)
   60   putStrLn "Symbol Table"
   61   putStrLn $ pretty (view symbols state)
   62   case res of
   63     Left err  -> print err
   64     Right ast -> debugPrint ast
   65 
   66 debugRepl :: IO ()

   72 -- Parsing utils for files
   73 readFileB :: String -> IO String
   74 readFileB fileName = do
   75   bts <- B.readFile fileName
   76   return (T.unpack . T.decodeUtf8 $ bts)
   77 
   78 safeReadFile :: String -> IO (Either String String)
   79 safeReadFile fileName = handle handleEx (Right <$> readFileB fileName)
   80   where handleEx :: IOException -> IO (Either String String)
   81         handleEx e = return (Left (show e))
   82 
   83 parseFile :: FilePath -> IO ()
   84 parseFile f = do
   85   s <- readFileB f
   86   let res = parse s
   87   case res of
   88     Left err -> print err
   89     Right p  -> print p
