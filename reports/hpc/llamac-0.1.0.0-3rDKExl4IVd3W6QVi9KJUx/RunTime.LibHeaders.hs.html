   78 initSymbolTable :: Parser ()
   79 initSymbolTable = do
   80   symbols . names %= openScope
   81   mapM_ f libSigs where
   82     f :: RunTimeLibSib -> Parser ()
   83     f (i, t, ps) = do
   84         let scheme = MonoType (constTypeToSymbolType t)
   85         argTs <- mapM genConstType (funToArgs ctCoAlg t)
   86         outT  <- genConstType (outFunType ctCoAlg t)
   87         fun   <- L.extern (L.mkName i) argTs outT
   88         symbols . names %= insert i (mkFullTableEntry (FunEntry scheme ps) (Just fun))
