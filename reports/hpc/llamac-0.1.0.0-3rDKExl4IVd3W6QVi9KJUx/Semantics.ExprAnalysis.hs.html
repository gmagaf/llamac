   28 {-
   29     Analyze let statements
   30     It will analyze all defs in the current typespace
   31     Generalize all function types
   32     Open a new namespace and write them in
   33 -}
   34 analyzeLet :: LetDef AlexPosn -> Parser (LetDef SemanticTag)
   35 analyzeLet (Let defs p) = do
   36     -- Analyze the signatures of the definitions
   37     sigAnalysisResults <- mapM analyzeDefSig defs
   38     -- Analyze the body of the definitions
   39     semDefEntries <- mapM analyzeDefBody sigAnalysisResults
   40     -- Get all the free variables from the outer scope and update them
   41     outerScopeVars <- getFreeTVars >>= resolveFreeVars
   42     -- Add free vars of mutables in outerScopeVars
   43     let mutVar (UntypedMutSig st _ _) acc   = tvarsInType st ++ acc
   44         mutVar (UntypedArrSig st _ _ _) acc = tvarsInType st ++ acc
   45         mutVar _ acc                        = acc
   46     let thisScopeTVars = foldr mutVar [] sigAnalysisResults
   47     let newFreeTVars = foldr S.insert outerScopeVars thisScopeTVars
   48     putFreeTVars newFreeTVars
   49     -- Generalize the results (using only outer scope tvars)
   50     finalRes <- mapM (genResult outerScopeVars) semDefEntries
   51     -- Open the scope
   52     openScopeInNames
   53     -- Insert the definitions in the scope
   54     mapM_ (uncurry insertName . B.first ide) finalRes
   55     -- Second analysis of the definitions to get the most general signatures
   56     let finalSemDefs = map fst finalRes
   57     return $ Let finalSemDefs (cpPosn p)
   58 analyzeLet (LetRec defs p) = do
   59     -- Analyze the signatures of the definitions
   60     sigAnalysisResults <- mapM analyzeDefSig defs
   61     -- Open the scope
   62     openScopeInNames
   63     -- Insert the definitions in the scope with placeholder type vars
   64     mapM_ (uncurry insertName . entryPair) sigAnalysisResults
   65     -- Analyze the body of the definitions
   66     semDefEntries <- mapM analyzeDefBody sigAnalysisResults
   67     -- Get all the free variables from the outer scope
   68     outerScopeVars <- getFreeTVars
   69     -- Add free vars of mutables in outerScopeVars
   70     let mutVar (UntypedMutSig st _ _) acc   = tvarsInType st ++ acc
   71         mutVar (UntypedArrSig st _ _ _) acc = tvarsInType st ++ acc
   72         mutVar _ acc                        = acc
   73     let thisScopeTVars = foldr mutVar [] sigAnalysisResults
   74     let newFreeTVars = foldr S.insert outerScopeVars thisScopeTVars
   75     -- Update the tvars in symbol table and in freeVars
   76     updatedEntries <- mapM (findName . ide . fst) semDefEntries
   77     freeVars <- resolveFreeVars newFreeTVars
   78     putFreeTVars freeVars
   79     -- Second analysis of the definitions to get the most general signatures
   80     let semDefs = map fst semDefEntries
   81     let genInput = zip3 semDefs defs updatedEntries
   82     res <- mapM secondAnalysis genInput
   83     -- Generalize the results
   84     finalRes <- mapM (genResult freeVars) res
   85     let finalSemDefs = map fst finalRes
   86     -- Final update in scope
   87     mapM_ (uncurry updateName . B.first ide) finalRes
   88     return $ LetRec finalSemDefs (cpPosn p)

   90 {-
   91     In Second analysis we only analyze Untyped fun definitions
   92     in order to get the most general unifier principal type
   93 -}
   94 secondAnalysis :: (Def SemanticTag, Def AlexPosn, TableEntry) -> Parser (Def SemanticTag, TableEntry)
   95 secondAnalysis (_, d@(FunDef {}), FunEntry _ _) = do
   96     sigRes <- analyzeDefSig d
   97     analyzeDefBody sigRes
   98 secondAnalysis (d, _, e) = return (d, e)

  100 {-
  101     We generalize second analysis results to

  104 genResult :: S.Set Int -> (Def SemanticTag, TableEntry) -> Parser (Def SemanticTag, TableEntry)
  105 genResult freeVars (FunDef i ps e tg, FunEntry (MonoType t) params) = do
  106     let scheme = gen freeVars t
  107     return (FunDef i ps e tg{typeInfo = DefType scheme}, FunEntry scheme params)
  108 genResult _ pair = return pair

  113 data SigAnalysisRes =
  114     TypedMutSig (Type SemanticTag) KeyEntryPair AlexPosn
  115   | TypedArrSig (Type SemanticTag) [Expr AlexPosn] KeyEntryPair AlexPosn
  116   | TypedFunSig (Type SemanticTag) [Param SemanticTag] (Expr AlexPosn) KeyEntryPair AlexPosn
  117   | UntypedMutSig SymbolType KeyEntryPair AlexPosn
  118   | UntypedArrSig SymbolType [Expr AlexPosn] KeyEntryPair AlexPosn
  119   | UnTypedFunSig SymbolType [Param SemanticTag] (Expr AlexPosn) KeyEntryPair AlexPosn

  121 entryPair :: SigAnalysisRes -> KeyEntryPair
  122 entryPair (TypedMutSig _ pair _)       = pair
  123 entryPair (TypedArrSig _ _ pair _)     = pair
  124 entryPair (TypedFunSig _ _ _ pair _)   = pair
  125 entryPair (UntypedMutSig _ pair _)     = pair
  126 entryPair (UntypedArrSig _ _ pair _)   = pair
  127 entryPair (UnTypedFunSig _ _ _ pair _) = pair

  132 {-
  133     Analysis of the signature of a definition
  134 -}
  135 analyzeDefSig :: Def AlexPosn -> Parser SigAnalysisRes
  136 analyzeDefSig (VarDef x p) = do
  137     tv <- freshTVar
  138     let varType = SymType . RefType $ tv
  139     return $ UntypedMutSig varType (x, MutableEntry varType) p
  140 analyzeDefSig (VarDefTyped x t p) = do
  141     semT <- stackTrace ("while analyzing mut var " ++ x) $ analyzeType t
  142     let varType = SymType . RefType $ typeToSymbolType semT
  143     return $ TypedMutSig semT (x, MutableEntry varType) p
  144 analyzeDefSig (ArrayDef i es p) = do
  145     let dims = length es
  146     tv <- freshTVar
  147     let arrayType = SymType . ArrayType dims $ tv
  148     return $ UntypedArrSig arrayType es (i, ArrayEntry arrayType dims) p
  149 analyzeDefSig (ArrayDefTyped i es t p) = do
  150     let dims = length es
  151     semT <- stackTrace ("while analyzing array " ++ i) $ analyzeType t
  152     let arrayType = SymType . ArrayType dims $ typeToSymbolType semT
  153     return $ TypedArrSig semT es (i, ArrayEntry arrayType dims) p
  154 analyzeDefSig (FunDef i ps e p) = do
  155     let paramNames = map ide ps
  156     when (hasDuplicates paramNames) $
  157         throwSemAtPosn ("Fun " ++ i ++ " cannot have duplicate params") p
  158     -- Analyze params in the current scope
  159     semPs <- mapM (stackTrace ("while analyzing the params of " ++ i) . analyzeParam) ps
  160     putSemPosn p
  161     paramTypes <- mapM getNodeType semPs
  162     -- Fresh outV is the output type of the function
  163     outV <- freshTVar
  164     let fType = paramsToFun SymType paramTypes outV
  165     -- fScheme <- gen fType
  166     return $ UnTypedFunSig fType semPs e (i, FunEntry (MonoType fType) paramNames) p
  167 analyzeDefSig (FunDefTyped i ps t e p) = do
  168     let paramNames = map ide ps
  169     -- First we make sure that there are no duplicates in params
  170     when (hasDuplicates paramNames) $
  171         throwSemAtPosn ("Fun " ++ i ++ " cannot have duplicate params") p
  172     -- We analyze the overall type of the function
  173     semT <- stackTrace ("while analyzing fun " ++ i) $ analyzeType t
  174     -- Analyze params and body in the current scope
  175     semPs <- mapM (stackTrace ("while analyzing the params of " ++ i) . analyzeParam) ps
  176     eT <- freshTVar
  177     -- Collect the results: the new param types, the expr type and unify tv with the result fun type
  178     putSemPosn p
  179     paramTypes <- mapM getNodeType semPs
  180     let fType = paramsToFun SymType paramTypes eT
  181     unify (typeToSymbolType semT, fType)
  182     let fScheme = MonoType $ typeToSymbolType semT
  183     return $ TypedFunSig semT semPs e (i, FunEntry fScheme paramNames) p

  201 {-
  202     Analysis of the body of a definition

  204 analyzeDefBody :: SigAnalysisRes -> Parser (Def SemanticTag, TableEntry)
  205 analyzeDefBody (UntypedMutSig st (i, entry) p) =
  206     let tg = SemTag{posn = p, typeInfo = DefType $ MonoType st}
  207     in return (VarDef i tg, entry)
  208 analyzeDefBody (TypedMutSig semT (i, entry) p) =
  209     let tg = SemTag{posn = p, typeInfo = DefType . MonoType . typeToSymbolType $ semT}
  210     in return (VarDefTyped i semT tg, entry)
  211 analyzeDefBody (UntypedArrSig st es (i, entry) p) = do
  212     semEs <- mapM (stackTrace ("while analyzing the dimensions of array " ++ i) . analyzeExpr) es
  213     typesEs <- mapM getNodeType semEs
  214     putSemPosn p
  215     mapM_ (unify . (,) (SymType IntType)) typesEs
  216     rSemEs <- mapM resolveNodeType semEs
  217     let tg = SemTag{posn = p, typeInfo = DefType $ MonoType st}
  218     return (ArrayDef i rSemEs tg, entry)
  219 analyzeDefBody (TypedArrSig semT es (i, entry) p) = do
  220     semEs <- mapM (stackTrace ("while analyzing the dimensions of array " ++ i) . analyzeExpr) es
  221     typesEs <- mapM getNodeType semEs
  222     putSemPosn p
  223     mapM_ (unify . (,) (SymType IntType)) typesEs
  224     rSemEs <- mapM resolveNodeType semEs
  225     let tg = SemTag{posn = p, typeInfo = DefType . MonoType . typeToSymbolType $ semT}
  226     return (ArrayDefTyped i rSemEs semT tg, entry)
  227 analyzeDefBody (UnTypedFunSig st semPs e (i, _) p) = do
  228     -- Hold the free vars outside the body
  229     outerScopeVars <- getFreeTVars
  230     -- Open scope for params names and their types
  231     openScopeInNames
  232     -- Insert params in the current scope
  233     let insertParam param = do
  234           t <- getNodeType param
  235           addFreeTVars t
  236           insertName (ide param) (ParamEntry t i)
  237           return t
  238     paramTypes <- mapM insertParam semPs
  239     semE <- stackTrace ("while analyzing the body of " ++ i) (analyzeExpr e)
  240     -- Collect the results: the new param types, the expr type and unify tv with the result fun type
  241     putSemPosn p
  242     eT <- getNodeType semE
  243     let fType = paramsToFun SymType paramTypes eT
  244     unify (st, fType)
  245     rSemPs <- mapM resolveNodeType semPs
  246     rSemE <- resolveNodeType semE
  247     reT <- getNodeType rSemE
  248     checkConstraint reT (NotAllowedFunType $ "Function " ++ i ++ " cannot return function type: " ++ pretty reT)
  249     rst <- resolveType st
  250     -- Close scope
  251     closeScopeInNames
  252     putFreeTVars outerScopeVars
  253     let fScheme = MonoType rst
  254     let tg = SemTag{posn = p, typeInfo = DefType fScheme}
  255     return (FunDef i rSemPs rSemE tg, FunEntry fScheme (map ide rSemPs))
  256 analyzeDefBody (TypedFunSig semT semPs e (i, _) p) = do
  257     -- Hold the free vars outside the body
  258     outerScopeVars <- getFreeTVars
  259     -- Open scope for params names and their types
  260     openScopeInNames
  261     -- Insert params in the current scope
  262     let insertParam param = do
  263           t <- getNodeType param
  264           addFreeTVars t
  265           insertName (ide param) (ParamEntry t i)
  266           return t
  267     paramTypes <- mapM insertParam semPs
  268     semE <- stackTrace ("while analyzing the body of " ++ i) (analyzeExpr e)
  269     -- Collect the results: the new param types, the expr type and unify tv with the result fun type
  270     putSemPosn p
  271     eT <- getNodeType semE
  272     let fType = paramsToFun SymType paramTypes eT
  273     let st = typeToSymbolType semT
  274     unify (st, fType)
  275     rSemPs <- mapM resolveNodeType semPs
  276     rSemE <- resolveNodeType semE
  277     reT <- getNodeType rSemE
  278     checkConstraint reT (NotAllowedFunType $ "Function " ++ i ++ " cannot return function type: " ++ pretty reT)
  279     -- Close scope
  280     closeScopeInNames
  281     putFreeTVars outerScopeVars
  282     let fScheme = MonoType st
  283     let tg = SemTag{posn = p, typeInfo = DefType fScheme}
  284     return (FunDefTyped i rSemPs semT rSemE tg, FunEntry fScheme (map ide rSemPs))

  288 analyzeExpr :: Expr AlexPosn -> Parser (Expr SemanticTag)

  291 -- Inductively analyze expressions using mapM for expression functor
  292 recSemExpr :: (ExprF (Expr SemanticTag) -> Parser (Expr SemanticTag))
  293          -> Expr AlexPosn
  294          -> Parser (Expr SemanticTag)
  295 recSemExpr g expr =
  296     let aexpr = stackTrace ("while analyzing expr " ++ pretty expr) . recSemExpr g
  297         atype = stackTrace ("while analyzing expr " ++ pretty expr) . analyzeType
  298         alet = stackTrace ("while analyzing expr " ++ pretty expr) . analyzeLet
  299         aclause = stackTrace ("while analyzing expr " ++ pretty expr) . analyzeClause
  300     in case expr of
  301         Expr (ForExpr i l u e) p -> do
  302             semL <- aexpr l
  303             semU <- aexpr u
  304             openScopeInNames
  305             insertName i (FunEntry (MonoType . SymType $ IntType) [])
  306             semE <- aexpr e
  307             closeScopeInNames
  308             putSemPosn p
  309             g (ForExpr i semL semU semE)
  310         Expr (ForDownExpr i u l e) p -> do
  311             semU <- aexpr u
  312             semL <- aexpr l
  313             openScopeInNames
  314             insertName i (FunEntry (MonoType . SymType $ IntType) [])
  315             semE <- aexpr e
  316             closeScopeInNames
  317             putSemPosn p
  318             g (ForDownExpr i semU semL semE)
  319         Expr ef p -> do
  320             semEf <- mapM aexpr ef
  321             putSemPosn p
  322             g semEf
  323         NewType t p -> do
  324             semT <- atype t
  325             putSemPosn p
  326             semNewType semT
  327         LetIn l e p -> do
  328             semL <- alet l
  329             semE <- aexpr e
  330             putSemPosn p
  331             semLetIn semL semE
  332         MatchExpr e cs p -> do
  333             semE <- aexpr e
  334             semCs <- mapM aclause cs
  335             putSemPosn p
  336             semMatchExpr semE semCs

  338 -- Util for returning the result expression by updating its node type
  339 retE :: ExprF (Expr SemanticTag) -> SymbolType -> Parser (Expr SemanticTag)
  340 retE ef t = do
  341     p <- getSemPosn
  342     return $ Expr ef SemTag{posn = p, typeInfo = NodeType t}

  344 indSemExpr :: ExprF (Expr SemanticTag) -> Parser (Expr SemanticTag)
  345 indSemExpr (IntCExpr c)           = retE (IntCExpr c) (SymType IntType)
  346 indSemExpr (FloatCExpr c)         = retE (FloatCExpr c) (SymType FloatType)
  347 indSemExpr (CharCExpr c)          = retE (CharCExpr c) (SymType CharType)
  348 indSemExpr (StringCExpr c)        = retE (StringCExpr c) (SymType (ArrayType 1 (SymType CharType)))
  349 indSemExpr TrueCExpr              = retE TrueCExpr (SymType BoolType)
  350 indSemExpr FalseCExpr             = retE FalseCExpr (SymType BoolType)
  351 indSemExpr UnitCExpr              = retE UnitCExpr (SymType UnitType)
  352 indSemExpr (ConstExpr i)          = semConstExpr i
  353 indSemExpr (ConstConstrExpr i)    = semConstConstrExpr i
  354 indSemExpr (FunAppExpr i es)      = semFunAppExpr i es
  355 indSemExpr (ConstrAppExpr i es)   = semConstrAppExpr i es
  356 indSemExpr (ArrayDim i dim)       = semArrayDim i dim
  357 indSemExpr (UnOpExpr op e)        = semUnOp op e
  358 indSemExpr (BinOpExpr op d e)     = semBinOp op d e
  359 indSemExpr (ArrayAccess i es)     = semArrayAccess i es
  360 indSemExpr (DeleteExpr e)         = semDeleteExpr e
  361 indSemExpr (BeginExpr e)          = getNodeType e >>= retE (BeginExpr e)
  362 indSemExpr (IfThenElseExpr c d e) = semIfThenElseExpr c d e
  363 indSemExpr (IfThenExpr c e)       = semIfThenExpr c e
  364 indSemExpr (WhileExpr c e)        = semWhileExpr c e
  365 indSemExpr (ForExpr i l u e)      = semForExpr i l u e
  366 indSemExpr (ForDownExpr i u l e)  = semForDownExpr i u l e

  371 semConstExpr :: Identifier -> Parser (Expr SemanticTag)
  372 semConstExpr i = do
  373     entry <- findName i
  374     case entry of
  375         FunEntry ft _ -> do
  376             t <- inst ft
  377             retE (ConstExpr i) t
  378         ParamEntry t _ -> do
  379             retE (ConstExpr i) t
  380         MutableEntry t -> do
  381             retE (ConstExpr i) t
  382         ArrayEntry t _ -> do
  383             retE (ConstExpr i) t
  384         PatternEntry t -> do
  385             retE (ConstExpr i) t
  386         ConstrEntry {} -> throwInternalError $
  387             "Constr entry: " ++ show entry ++ " is not expected for identifier key " ++ i

  389 semConstConstrExpr :: ConstrIdentifier -> Parser (Expr SemanticTag)

  394         _                 -> throwInternalError $

  397 semFunAppExpr :: Identifier -> [Expr SemanticTag] -> Parser (Expr SemanticTag)
  398 semFunAppExpr i es = let funTypeToArgTypes = funToArgs stCoAlg in do
  399     entry <- findName i
  400     case entry of
  401         FunEntry ft ps ->
  402             case compare (length es) (length ps) of
  403                 LT -> throwSem $ "Function " ++ i ++ " is applied to too few arguments"
  404                 GT -> throwSem $ "Function " ++ i ++ " is applied to too many arguments"
  405                 EQ -> do
  406                     ts <- mapM getNodeType es
  407                     v <- freshTVar
  408                     t <- inst ft
  409                     let inf = paramsToFun SymType ts v
  410                     unify (t, inf)
  411                     res <- mapM resolveNodeType es
  412                     rv <- resolveType v
  413                     checkConstraint rv (NotAllowedFunType $ "Function " ++ i ++ " cannot return function type: " ++ pretty rv)
  414                     retE (FunAppExpr i res) rv
  415         ParamEntry t _ -> do
  416             ts <- mapM getNodeType es
  417             v <- freshTVar
  418             let inf = paramsToFun SymType ts v
  419             unify (t, inf)
  420             res <- mapM resolveNodeType es
  421             rv <- resolveType v
  422             rt <- resolveType t
  423             let argTypes = funTypeToArgTypes rt
  424             case compare (length es) (length argTypes) of
  425                 LT -> throwSem $ "Param " ++ i ++ " of type " ++ pretty t ++ " is applied to too few arguments"
  426                 GT -> throwSem $ "Param " ++ i ++ " of type " ++ pretty t ++ " is applied to too many arguments"
  427                 EQ -> retE (FunAppExpr i res) rv
  428         PatternEntry t -> do
  429             ts <- mapM getNodeType es
  430             v <- freshTVar
  431             let inf = paramsToFun SymType ts v
  432             unify (t, inf)
  433             res <- mapM resolveNodeType es
  434             rv <- resolveType v
  435             rt <- resolveType t
  436             let argTypes = funTypeToArgTypes rt
  437             case compare (length es) (length argTypes) of
  438                 LT -> throwSem $ "Pattern " ++ i ++ " of type " ++ pretty t ++ " is applied to too few arguments"
  439                 GT -> throwSem $ "Pattern " ++ i ++ " of type " ++ pretty t ++ " is applied to too many arguments"
  440                 EQ -> retE (FunAppExpr i res) rv
  441         MutableEntry _    -> throwSem $ "Cannot apply arguments to the mutable variable " ++ i
  442         ArrayEntry _ _    -> throwSem $ "Cannot apply arguments to array " ++ i
  443         ConstrEntry {}    -> throwSem $ "Cannot apply function arguments to constr " ++ i
  444 
  445 semConstrAppExpr :: ConstrIdentifier -> [Expr SemanticTag] -> Parser (Expr SemanticTag)
  446 semConstrAppExpr i es = do
  447     entry <- findName i
  448     case entry of
  449         ConstrEntry t psT _ ->
  450             case compare (length es) (length psT) of
  451                 LT -> throwSem $ "Constructor " ++ i ++ " is applied to too few arguments"
  452                 GT -> throwSem $ "Constructor " ++ i ++ " is applied to too many arguments"
  453                 EQ -> do
  454                     ts <- mapM getNodeType es
  455                     v <- freshTVar
  456                     let inf = paramsToFun SymType ts v
  457                     unify (constTypeToSymbolType t, inf)
  458                     res <- mapM resolveNodeType es
  459                     rv <- resolveType v
  460                     checkConstraint rv (NotAllowedFunType $ "Constructor " ++ i ++ " cannot return function type: " ++ pretty rv)
  461                     retE (ConstrAppExpr i res) rv
  462         _ -> throwInternalError $
  463             "Entry: " ++ show entry ++ " is not expected for constructor identifier key " ++ i

  465 semArrayDim :: Identifier -> Int -> Parser (Expr SemanticTag)
  466 semArrayDim i dim | dim < 1 = throwSem $ "Cannot compute the " ++ show dim ++ " dimension of array " ++ i
  467 semArrayDim i dim = findName i >>= run where
  468     run (ArrayEntry _ dims)
  469       | dims < dim = throwSem $ "Cannot compute the " ++ show dim ++ " dimension of " ++ show dims ++ "-dim array " ++ i
  470       | otherwise = retE (ArrayDim i dim) (SymType IntType)
  471     run (ParamEntry t _) = do
  472         checkConstraint t (ArrayOfAtLeastDim dim $ "Cannot compute the dimension " ++ show dim ++ " for type " ++ pretty t)
  473         retE (ArrayDim i dim) (SymType IntType)
  474     run (FunEntry sch []) = do
  475         t <- inst sch
  476         checkConstraint t (ArrayOfAtLeastDim dim $ "Cannot compute the dimension " ++ show dim ++ " for type " ++ pretty t)
  477         retE (ArrayDim i dim) (SymType IntType)
  478     run (PatternEntry t) = do
  479         checkConstraint t (ArrayOfAtLeastDim dim $ "Cannot compute the dimension " ++ show dim ++ " for type " ++ pretty t)
  480         retE (ArrayDim i dim) (SymType IntType)
  481     run _ = throwSem $ "No array " ++ i ++ " in scope"

  490 semUnOp :: UnOp -> Expr SemanticTag -> Parser (Expr SemanticTag)
  491 semUnOp op e = do
  492     t <- getNodeType e
  493     v <- freshTVar
  494     case op of
  495         BangOp         -> unify (SymType (RefType v), t)
  496         PlusUnOp       -> unify (SymType IntType, t)
  497         MinusUnOp      -> unify (SymType IntType, t)
  498         PlusFloatUnOp  -> unify (SymType FloatType, t)
  499         MinusFloatUnOp -> unify (SymType FloatType, t)
  500         NotOp          -> unify (SymType BoolType, t)
  501     rE <- resolveNodeType e
  502     rt <- resolveType t
  503     let finalT = if op == BangOp then v else rt
  504     retE (UnOpExpr op rE) finalT
  505 
  506 semBinOp :: BinOp -> Expr SemanticTag -> Expr SemanticTag -> Parser (Expr SemanticTag)
  507 semBinOp op d e = do
  508     s <- getNodeType d
  509     t <- getNodeType e
  510     outT <- freshTVar
  511     case op of
  512         PlusOp          -> unifyAll (SymType IntType) s t outT
  513         MinusOp         -> unifyAll (SymType IntType) s t outT
  514         TimesOp         -> unifyAll (SymType IntType) s t outT
  515         DivOp           -> unifyAll (SymType IntType) s t outT
  516         ModOp           -> unifyAll (SymType IntType) s t outT
  517         PlusFloatOp     -> unifyAll (SymType FloatType) s t outT
  518         MinusFloatOp    -> unifyAll (SymType FloatType) s t outT
  519         TimesFloatOp    -> unifyAll (SymType FloatType) s t outT
  520         DivFloatOp      -> unifyAll (SymType FloatType) s t outT
  521         ExpOp           -> unifyAll (SymType FloatType) s t outT
  522         AndOp           -> unifyAll (SymType BoolType) s t outT
  523         OrOp            -> unifyAll (SymType BoolType) s t outT
  524         SemicolonOp     -> unify (t, outT)
  525         AssignMutableOp -> do
  526             unify (s, SymType (RefType t))
  527             unify (SymType UnitType, outT)
  528         EqOp            -> unifyEq s t outT
  529         NotEqOp         -> unifyEq s t outT
  530         NatEqOp         -> unifyEq s t outT
  531         NotNatEqOp      -> unifyEq s t outT
  532         LTOp            -> unifyComp s t outT
  533         GTOp            -> unifyComp s t outT
  534         LEqOp           -> unifyComp s t outT
  535         GEqOp           -> unifyComp s t outT
  536     rd <- resolveNodeType d
  537     re <- resolveNodeType e
  538     retE (BinOpExpr op rd re) outT where
  539         unifyAll expected s' t' outT' = do
  540             unify (expected, s')
  541             unify (expected, t')
  542             unify (expected, outT')
  543         unifyEq s' t' outT' = do
  544             checkConstraint s' (NotAllowedFunType $ "Cannot apply operator " ++ pretty op ++ " to fun types")
  545             checkConstraint s' (NotAllowedArrayType $ "Cannot apply operator " ++ pretty op ++ " to array types")
  546             checkConstraint t' (NotAllowedFunType $ "Cannot apply operator " ++ pretty op ++ " to fun types")
  547             checkConstraint t' (NotAllowedArrayType $ "Cannot apply operator " ++ pretty op ++ " to array types")
  548             unify (s', t')
  549             unify (SymType BoolType, outT')
  550         unifyComp s' t' outT' = do
  551             let c = mkAllowedTypes [ConstType IntType, ConstType FloatType, ConstType CharType]
  552                     ("Operator " ++ pretty op ++ " can only be applied to int, float or char")
  553             checkConstraint s' c
  554             checkConstraint t' c
  555             unify (s', t')
  556             unify (SymType BoolType, outT')
  557 
  558 semArrayAccess :: Identifier -> [Expr SemanticTag] -> Parser (Expr SemanticTag)
  559 semArrayAccess i es = do
  560     entry <- findName i
  561     case entry of
  562         ArrayEntry t dims ->
  563             case compare (length es) dims of
  564                 LT -> throwSem $ "Array " ++ i ++ " is applied to too few arguments"
  565                 GT -> throwSem $ "Array " ++ i ++ " is applied to too many arguments"
  566                 EQ -> do
  567                     ts <- mapM getNodeType es
  568                     mapM_ (\et -> unify (SymType IntType, et)) ts
  569                     v <- freshTVar
  570                     let inf = SymType (ArrayType dims v)
  571                     unify (t, inf)
  572                     res <- mapM resolveNodeType es
  573                     rv <- resolveType v
  574                     retE (ArrayAccess i res) (SymType (RefType rv))
  575         ParamEntry t _ -> do
  576             ts <- mapM getNodeType es
  577             mapM_ (\et -> unify (SymType IntType, et)) ts
  578             v <- freshTVar
  579             let inf = SymType (ArrayType (length es) v)
  580             unify (t, inf)
  581             res <- mapM resolveNodeType es
  582             rv <- resolveType v
  583             retE (ArrayAccess i res) (SymType (RefType rv))
  584         FunEntry s []  -> do
  585             ts <- mapM getNodeType es
  586             mapM_ (\et -> unify (SymType IntType, et)) ts
  587             t <- inst s
  588             v <- freshTVar
  589             let inf = SymType (ArrayType (length es) v)
  590             unify (t, inf)
  591             res <- mapM resolveNodeType es
  592             rv <- resolveType v
  593             retE (ArrayAccess i res) (SymType (RefType rv))
  594         PatternEntry t -> do
  595             ts <- mapM getNodeType es
  596             mapM_ (\et -> unify (SymType IntType, et)) ts
  597             v <- freshTVar
  598             let inf = SymType (ArrayType (length es) v)
  599             unify (t, inf)
  600             res <- mapM resolveNodeType es
  601             rv <- resolveType v
  602             retE (ArrayAccess i res) (SymType (RefType rv))
  603         _    -> throwSem $ "No array " ++ i ++ " found in scope"
  604

  694 semMatchExpr :: Expr SemanticTag
  695              -> [Clause SemanticTag]
  696              -> Parser (Expr SemanticTag)
  697 semMatchExpr e cs = do
  698     p <- getSemPosn
  699     et <- getNodeType e
  700     checkConstraint et (AllowedUserDefinedType "Can only apply pattern matching to user defined type")
  701     let getPat (Match pat _ _) = pat
  702     let pats = map getPat cs
  703     patTs <- mapM getNodeType pats
  704     mapM_ (\t -> unify (et, t)) patTs
  705     outT <- freshTVar
  706     let getExp (Match _ expr _) = expr
  707     let patExps = map getExp cs
  708     patExpTs <- mapM getNodeType patExps
  709     mapM_ (\t -> unify (outT, t)) patExpTs
  710     re <- resolveNodeType e
  711     rcs <- mapM resClause cs
  712     routT <- resolveType outT
  713     return $ MatchExpr re rcs SemTag{posn = p, typeInfo = NodeType routT} where
  714         resClause (Match pat expr t) = do
  715             rPat <- resolveNodeType pat
  716             rExp <- resolveNodeType expr
  717             return (Match rPat rExp t)

  719 -- Semantic analysis of clauses
  720 
  721 analyzeClause :: Clause AlexPosn -> Parser (Clause SemanticTag)
  722 analyzeClause c@(Match pat e p) = do
  723     -- Hold the free vars outside the body
  724     outerScopeVars <- getFreeTVars
  725     openScopeInNames
  726     semP <- stackTrace ("while analyzing clause " ++ pretty c) $ analyzePattern pat
  727     semE <- stackTrace ("while analyzing clause " ++ pretty c) $ analyzeExpr e
  728     closeScopeInNames
  729     -- Restore the free tVars
  730     putFreeTVars outerScopeVars
  731     return $ Match semP semE (cpPosn p)

  733 -- Semantic analysis of patterns
  734 
  735 analyzePattern :: Pattern AlexPosn -> Parser (Pattern SemanticTag)
  736 analyzePattern = recSemPattern indSemPat

  738 recSemPattern :: (PatternF (Pattern SemanticTag) -> Parser (Pattern SemanticTag))
  739     -> Pattern AlexPosn
  740     -> Parser (Pattern SemanticTag)
  741 recSemPattern f p@(Pattern pf psn) = do
  742     let aPattern = stackTrace ("while analyzing pattern " ++ pretty p) . recSemPattern f
  743     semPf <- mapM aPattern pf
  744     putSemPosn psn
  745     f semPf

  747 retP :: PatternF (Pattern SemanticTag) -> SymbolType -> Parser (Pattern SemanticTag)
  748 retP pat t = do
  749     p <- getSemPosn
  750     return $ Pattern pat SemTag{posn = p, typeInfo = NodeType t}

  752 indSemPat :: PatternF (Pattern SemanticTag) -> Parser (Pattern SemanticTag)
  753 indSemPat p@(IntConstPattern {}) = retP p (SymType IntType)
  754 indSemPat p@(FloatConstPattern {}) = retP p (SymType FloatType)
  755 indSemPat p@(CharConstPattern {}) = retP p (SymType CharType)
  756 indSemPat TruePattern = retP TruePattern (SymType BoolType)
  757 indSemPat FalsePattern = retP FalsePattern (SymType BoolType)
  758 indSemPat (IdPattern x) = do
  759     v <- freshTVar
  760     addFreeTVars v
  761     insertName x (PatternEntry v)
  762     retP (IdPattern x) v
  763 indSemPat (ConstrPattern i pats) = do
  764     entry <- findName i
  765     case entry of
  766         ConstrEntry _ argT outT ->
  767             case compare (length pats) (length argT) of
  768                 LT -> throwSem $ "Constructor " ++ i ++ " is applied to too few patterns"
  769                 GT -> throwSem $ "Constructor " ++ i ++ " is applied to too many patterns"
  770                 EQ -> do
  771                     patTs <- mapM getNodeType pats
  772                     mapM_ unify (zipWith (\ct t -> (constTypeToSymbolType ct, t)) argT patTs)
  773                     rpats <- mapM resolveNodeType pats
  774                     mapM_ verifyParamPat rpats
  775                     retP (ConstrPattern i rpats) (constTypeToSymbolType outT)
  776             where verifyParamPat (Pattern (ConstrPattern _ []) _) = return ()
  777                   verifyParamPat (Pattern (ConstrPattern p _) tg) =
  778                     throwSemAtPosn ("Pattern param " ++ p ++
  779                         " cannot be a pattern of a constructor with parameters") (posn tg)
  780                   verifyParamPat _ = return ()
  781         _ -> throwInternalError $
  782             "Entry: " ++ show entry ++ " is not expected for constructor identifier key " ++ i
