   15 data SemanticState = SemanticState
   16   { varTypeC     :: Int         -- a counter to the var types used
   17   , posnOfSem    :: AlexPosn    -- a posn to the current place of analysis, used for error messages
   18   , unifier      :: Unifier     -- this should always be the most
   19                                 -- general unifier and it should keep track
   20                                 -- of variables not in scope. This should work
   21                                 -- if all the fresh variables have a new name.
   22                                 -- This is ensured by the varTypeC. However, the
   23                                 -- result for each type in scope should be
   24                                 -- a type in scope.
   25   , constraints  :: ConstraintsMap
   26                                 -- a map to hold all the constraints for each type var
   27   , freeTVars    :: S.Set Int   -- a set of all free type variables at current scope
   28   }
   29 
   30 instance Show SemanticState where
   31   show s = "SemanticState {" ++
   32     "varTypeC = " ++ show (varTypeC s)
   33     ++ ", " ++ "posnOfSem = " ++ show (posnOfSem s)
   34     ++ ", " ++ "unifier = " ++ showUnifierVals (unifier s) (varTypeC s)
   35     ++ ", " ++ "constraints = " ++ showConstraintsMap (constraints s)
   36     ++ ", " ++ "freeTVarsInScope = " ++ showFreeTVars (freeTVars s)
   37     ++ "}"
   38 
   39 showUnifierVals :: Unifier -> Int -> String
   40 showUnifierVals _ 0 = "_"
   41 showUnifierVals f n = intercalate ", " $ map (showUnifierVal f . TVar) [0..(n - 1)]
   42 
   43 showUnifierVal :: Unifier -> SymbolType -> String
   44 showUnifierVal f st = "U(" ++ pretty st ++ ") = " ++ val where
   45   val = maybe "Nothing" pretty (f st)
   46 
   47 showFreeTVars :: S.Set Int -> String
   48 showFreeTVars s = "[" ++ intercalate ", " (map (pretty . TVar) (S.toList s)) ++ "]"
   49 
   50 initSemanticState :: SemanticState
   51 initSemanticState = SemanticState
   52                 {
   53                   varTypeC = 0
   54                 , posnOfSem = alexStartPos
   55                 , unifier = initUnifier
   56                 , constraints = emptyConstraintsMap
   57                 , freeTVars = S.empty
   58                 }
   59 
   60 initUnifier :: SymbolType -> Maybe SymbolType
   61 initUnifier (SymType tf) = SymType <$> mapM initUnifier tf
   62 initUnifier (TVar _) = Nothing
