   20 -- This module contains the semantic analysis of the nodes
   21 -- and decorates them with the semantic tag
   22 
   23 -- Organize the semantic analyzable nodes in a class
   24 class Node f => Analyzable f where
   25     sem :: f AlexPosn -> Parser (f SemanticTag)
   26     semTag :: f AlexPosn -> Parser SemanticTag
   27     semTag f = tag <$> sem f
   28 
   29 class Analyzable f => TypeAble f where
   30     infer :: f AlexPosn -> Parser SymbolType
   31     infer f = do
   32         tg <- semTag f
   33         case typeInfo tg of
   34             NodeType t -> return t
   35             _          -> throwSemAtPosn "Could not infer type of expr" (posn tg)
   36     typeCheck :: f AlexPosn -> SymbolType -> Parser Bool
   37     typeCheck f t = (t ==) <$> infer f
   38 
   39 -- Functions for analyzing nodes
   40 analyzeAST :: AST AlexPosn -> Parser (AST SemanticTag)
   41 analyzeAST ast = do
   42     semAst <- mapM (bitraverse sem sem) ast
   43     overNamesM $ mapM (\fullEntry -> do
   44                 e <- resolveTableEntry (view basicInfo fullEntry)
   45                 return (set basicInfo e fullEntry))
   46     mapM (bitraverse resolveNodeRec return) semAst
   47 
   48 instance Analyzable TypeDef where

   50 instance Analyzable Type where
