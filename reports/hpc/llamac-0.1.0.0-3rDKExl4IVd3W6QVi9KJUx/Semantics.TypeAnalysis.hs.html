   32 analyzeTypeDef :: TypeDef AlexPosn -> Parser (TypeDef SemanticTag)
   33 analyzeTypeDef (TypeDef tDefs p) =
   34     let typeNames = S.fromList $ map ide tDefs
   35     in do
   36         openScopeInTypes
   37         openScopeInNames
   38         mapM_ (insertTypeDef typeNames) tDefs
   39         semTDefs <- mapM analyzeTDef tDefs
   40         return $ TypeDef semTDefs (cpPosn p)
   41 
   42 insertTypeDef :: S.Set Identifier -> TDef AlexPosn -> Parser ()
   43 insertTypeDef typesInDef (TDef tId cs p) =
   44     let constrNames :: [ConstrIdentifier]
   45         constrNames = map ide cs
   46         checkDuplicateConstrs :: Parser ()
   47         checkDuplicateConstrs = when (hasDuplicates constrNames) $
   48             throwSem $ "Type " ++ tId ++ " cannot have duplicate constructors"
   49         checkTypeInCtx :: Type AlexPosn -> Parser ConstType
   50         checkTypeInCtx t@(Type (UserDefinedType tName) tp) =
   51             if S.member tName typesInDef
   52             then return (typeTo ConstType t)
   53             else do
   54                 putSemPosn tp
   55                 checkTypeInScope tName
   56                 return (typeTo ConstType t)
   57         checkTypeInCtx t = return (typeTo ConstType t)
   58         checkConstrParams :: Constr AlexPosn -> Parser (ConstrIdentifier, [ConstType])
   59         checkConstrParams (Constr c ts _) = do
   60             checkedTs <- mapM checkTypeInCtx ts
   61             return (c, checkedTs)
   62     in do
   63         putSemPosn p
   64         checkDuplicateConstrs
   65         constrs <- mapM checkConstrParams cs
   66         insertType tId (TypeEntry constrs)

   68 analyzeTDef :: TDef AlexPosn -> Parser (TDef SemanticTag)
   69 analyzeTDef (TDef tId cs p) = do
   70     semCs <- mapM (analyzeConstr tId) cs
   71     return $ TDef tId semCs (cpPosn p)
   72 
   73 analyzeConstr :: Identifier -> Constr AlexPosn -> Parser (Constr SemanticTag)
   74 analyzeConstr tId (Constr cId params p) = do
   75     semParams <- mapM (stackTrace ("while analyzing constr " ++ cId) . analyzeType) params
   76     let paramTypes = map (typeTo ConstType) semParams
   77     let outputType = ConstType $ UserDefinedType tId
   78     let typeOfConstr = paramsToFun ConstType paramTypes outputType
   79     insertName cId (ConstrEntry typeOfConstr paramTypes outputType)
   80     return $ Constr cId semParams (cpPosn p)

   82 recSemType :: (TypeF (Type SemanticTag) -> Parser (Type SemanticTag))
   83     -> Type AlexPosn
   84     -> Parser (Type SemanticTag)
   85 recSemType f t@(Type tf p) = do
   86     let aType = stackTrace ("while analyzing type " ++ pretty t) . recSemType f
   87     semTf <- mapM aType tf
   88     putSemPosn p
   89     f semTf

   91 analyzeType :: Type AlexPosn -> Parser (Type SemanticTag)
   92 analyzeType = recSemType aType where
   93     aType :: TypeF (Type SemanticTag) -> Parser (Type SemanticTag)
   94     aType (ArrayType dim _) | dim < 1 = throwSem "Dimension of array type can't be less than 1"
   95     aType (UserDefinedType t) = do
   96         checkTypeInScope t
   97         Type (UserDefinedType t) . cpPosn <$> getSemPosn
   98     aType tf = Type tf . cpPosn <$> getSemPosn
