   30 data TypeConstraintK = AllowedTypesT
   31                      | NotAllowedFunTypeT
   32                      | NotAllowedArrayTypeT
   33                      | ArrayOfAtLeastDimT
   34                      | AllowedUserDefinedTypeT
   35     deriving Show
   36 
   37 data Tag (tg :: TypeConstraintK) where
   38     AllowedTypesTg           :: Tag 'AllowedTypesT
   39     NotAllowedFunTypeTg      :: Tag 'NotAllowedFunTypeT
   40     NotAllowedArrayTypeTg    :: Tag 'NotAllowedArrayTypeT
   41     ArrayOfAtLeastDimTg      :: Tag 'ArrayOfAtLeastDimT
   42     AllowedUserDefinedTypeTg :: Tag 'AllowedUserDefinedTypeT
   43 
   44 deriving instance Show (Tag a)
   45 deriveGEq      ''Tag
   46 deriveArgDict  ''Tag
   47 deriveGCompare ''Tag
   48 deriveGShow    ''Tag
   49 
   50 data TypeConstraint (tg :: TypeConstraintK) where
   51     AllowedTypes           :: S.Set ConstType -> String -> TypeConstraint 'AllowedTypesT
   52     NotAllowedFunType      :: String -> TypeConstraint 'NotAllowedFunTypeT
   53     NotAllowedArrayType    :: String -> TypeConstraint 'NotAllowedArrayTypeT
   54     ArrayOfAtLeastDim      :: Int -> String -> TypeConstraint 'ArrayOfAtLeastDimT
   55     AllowedUserDefinedType :: String -> TypeConstraint 'AllowedUserDefinedTypeT
   56 
   57 deriving instance Show (TypeConstraint a)
   58 
   59 showDPair :: DS.DSum Tag TypeConstraint -> String
   60 showDPair (_ DS.:=> c) = show c
   61 
   62 mkAllowedTypes :: [ConstType] -> String -> TypeConstraint 'AllowedTypesT
   63 mkAllowedTypes types = AllowedTypes (S.fromList types)
   64 
   65 newtype ConstraintSet = ConstraintSet { getMap :: DM.DMap Tag TypeConstraint }
   66 
   67 instance Show ConstraintSet where
   68     show (ConstraintSet m) = "{" ++ fst (DM.foldrWithKey showVal ("}", True) m)
   69         where showVal _ v (acc, first) = (show v ++ (if first then "" else ", ") ++ acc, False)
   70 
   71 emptyConstraintSet :: ConstraintSet
   72 emptyConstraintSet = ConstraintSet DM.empty
   73 
   74 getTag :: forall (c :: TypeConstraintK). TypeConstraint c -> Tag c
   75 getTag (AllowedTypes {})           = AllowedTypesTg
   76 getTag (NotAllowedFunType {})      = NotAllowedFunTypeTg
   77 getTag (NotAllowedArrayType {})    = NotAllowedArrayTypeTg
   78 getTag (ArrayOfAtLeastDim {})      = ArrayOfAtLeastDimTg
   79 getTag (AllowedUserDefinedType {}) = AllowedUserDefinedTypeTg
   80 
   81 singletonSet :: forall (c :: TypeConstraintK). TypeConstraint c -> ConstraintSet
   82 singletonSet tc = ConstraintSet $ DM.singleton (getTag tc) tc
   83 
   84 combineConstrs :: forall v. Tag v -> TypeConstraint v -> TypeConstraint v -> TypeConstraint v
   85 combineConstrs AllowedTypesTg (AllowedTypes s1 e1) (AllowedTypes s2 _)                             = AllowedTypes (S.intersection s1 s2) e1
   86 combineConstrs NotAllowedFunTypeTg (NotAllowedFunType s) (NotAllowedFunType _)                     = NotAllowedFunType s
   87 combineConstrs NotAllowedArrayTypeTg (NotAllowedArrayType s) (NotAllowedArrayType _)               = NotAllowedArrayType s
   88 combineConstrs ArrayOfAtLeastDimTg (ArrayOfAtLeastDim d1 s1) (ArrayOfAtLeastDim d2 s2) | d1 < d2   = ArrayOfAtLeastDim d2 s2
   89                                                                                        | otherwise = ArrayOfAtLeastDim d1 s1
   90 combineConstrs AllowedUserDefinedTypeTg (AllowedUserDefinedType s) (AllowedUserDefinedType _)      = AllowedUserDefinedType s
   91 
   92 union :: ConstraintSet -> ConstraintSet -> ConstraintSet
   93 union s1 s2 = ConstraintSet $ DM.unionWithKey combineConstrs (getMap s1) (getMap s2)
   94 
   95 traverse :: Applicative t => (forall c. Tag c -> TypeConstraint c -> t ()) -> ConstraintSet -> t ()
   96 traverse f = DM.traverseWithKey_ f . getMap
   97 
   98 type ConstraintsMap = IM.IntMap ConstraintSet
   99 
  100 showConstraintsMap :: ConstraintsMap -> String
  101 showConstraintsMap m = "[" ++ fst (IM.foldrWithKey showVal ("]", True) m)
  102         where showVal k v (acc, first) = (pretty (TVar k) ++ ": " ++ show v ++ (if first then "" else ", ") ++ acc, False)
  103 
  104 emptyConstraintsMap :: ConstraintsMap
  105 emptyConstraintsMap = IM.empty
  106 
  107 lookupConstr :: Int -> ConstraintsMap -> Maybe ConstraintSet
  108 lookupConstr = IM.lookup
  109 
  110 insertConstr :: Int -> ConstraintSet -> ConstraintsMap -> ConstraintsMap
  111 insertConstr = IM.insert
  112 
  113 insertConstrWith :: (ConstraintSet -> ConstraintSet -> ConstraintSet) -> Int -> ConstraintSet -> ConstraintsMap -> ConstraintsMap
  114 insertConstrWith = IM.insertWith
