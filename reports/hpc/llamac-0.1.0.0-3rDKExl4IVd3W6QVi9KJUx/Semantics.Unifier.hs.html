   20 checkConstraint :: SymbolType -> TypeConstraint t -> Parser ()
   21 checkConstraint (SymType (FunType _ _)) (NotAllowedFunType s) =
   22     throwSem $ "Type constraint failed: " ++ s
   23 checkConstraint (SymType _) (NotAllowedFunType _) = return ()
   24 checkConstraint (SymType (ArrayType _ _)) (NotAllowedArrayType s) =
   25     throwSem $ "Type constraint failed: " ++ s
   26 checkConstraint (SymType _) (NotAllowedArrayType _) = return ()
   27 checkConstraint (SymType (ArrayType d _)) (ArrayOfAtLeastDim l s) =
   28     when (d < l) $ throwSem $ "Type constraint failed: " ++ s
   29 checkConstraint (SymType _) (ArrayOfAtLeastDim _ s) = throwSem $ "Type constraint failed: " ++ s
   30 checkConstraint t@(SymType _) (AllowedTypes ts s) =
   31     let eqTypes = any (\ct -> t == constTypeToSymbolType ct) ts
   32     in unless eqTypes . throwSem $ "Type constraint failed for type " ++ pretty t ++ ": " ++ s
   33 checkConstraint (SymType (UserDefinedType _)) (AllowedUserDefinedType _) = return ()
   34 checkConstraint t@(SymType _) (AllowedUserDefinedType s) =
   35     throwSem $ "Type constraint failed: for type " ++ pretty t ++ ". " ++ s
   36 checkConstraint (TVar v) c = do
   37     let tv = TVar v
   38     f <- getUnifier
   39     when (isNothing (f tv)) $
   40         throwSem ("Unable to add constraint: " ++ show c ++
   41                   " . Variable " ++ pretty tv ++ " has never been used before")
   42     cs <- getConstraints
   43     putConstraints $ insertConstrWith union v (singletonSet c) cs
   44 
   45 applyConstraints :: (SymbolType, SymbolType) -> Parser ()
   46 applyConstraints (TVar v, TVar u) = copyConstraints (TVar v, TVar u)
   47 applyConstraints (TVar v, t) = do
   48     c <- getConstraints
   49     forM_ (lookupConstr v c) (traverse (\ _ -> checkConstraint t))
   50 applyConstraints (t, TVar v) = do
   51     c <- getConstraints
   52     forM_ (lookupConstr v c) (traverse (\ _ -> checkConstraint t))
   53 applyConstraints _ = return ()
   54 
   55 -- This function is used to unify two types
   56 -- Given a type constraint it extends the unifier
   57 -- by resolving the constraint.
   58 -- The constraint is solved in favor of the less
   59 -- variable if possible
   60 -- The result is saved in SymbolTable
   61 unify :: (SymbolType, SymbolType) -> Parser ()
   62 unify (st1, st2) = do
   63     rt <- resolveType st1
   64     rs <- resolveType st2
   65     applyConstraints (rt, rs)
   66     case (rt, rs) of
   67         (t, s) | t == s -> return ()
   68         (TVar v, TVar u) | v < u -> unify (rs, rt)
   69         (TVar v, s) | notVarInType v s -> putUnifier v s
   70         (t, TVar v) | notVarInType v t -> putUnifier v t
   71         (SymType (FunType t1 t2), SymType (FunType s1 s2)) -> do
   72             unify (t1, s1)
   73             unify (t2, s2)
   74         (SymType (ArrayType dimT t), SymType (ArrayType dimS s)) | dimT == dimS -> do
   75             unify (t, s)
   76         (SymType (RefType t), SymType (RefType s)) -> do
   77             unify (t, s)
   78         _ -> throwSem $ "Unable to unify type " ++ pretty rt ++ " with " ++ pretty rs
