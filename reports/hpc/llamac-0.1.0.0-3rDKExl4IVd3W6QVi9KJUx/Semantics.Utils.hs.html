   69 getUnifier :: Parser Unifier
   70 getUnifier = unifier <$> getSemState

   72 addUnifier :: Int -> Parser ()
   73 addUnifier v = do
   74     let tv = TVar v
   75     f <- getUnifier
   76     when (isJust (f tv)) $
   77         throwSem ("Fresh variable " ++ pretty tv ++ " has been used before")
   78     let g t@(TVar _)   = if t == tv then Just tv else f t
   79         g (SymType tf) = SymType <$> mapM g tf
   80     s <- getSemState
   81     putSemState s{unifier = g}
   82 
   83 putUnifier :: Int -> SymbolType -> Parser ()
   84 putUnifier v t = do
   85     let tv = TVar v
   86     f <- getUnifier
   87     when (isNothing (f tv)) $
   88         throwSem ("Variable " ++ pretty tv ++ " has never been used before")
   89     when (isNothing (f t)) $
   90         throwSem ("Type " ++ pretty t ++ " contains variables never used before")
   91     let g t'@(TVar v') = if v' == v then Just t else Just t'
   92         g (SymType tf) = SymType <$> mapM g tf
   93     s <- getSemState
   94     putSemState s{unifier = f >=> g}

  118 copyConstraints :: (SymbolType, SymbolType) -> Parser ()
  119 copyConstraints (TVar v, TVar u) = do
  120     c <- getConstraints
  121     case (lookupConstr v c, lookupConstr u c) of
  122         (Nothing, Nothing) -> return ()
  123         (Nothing, Just cs) -> putConstraints $ insertConstr v cs c
  124         (Just cs, Nothing) -> putConstraints $ insertConstr u cs c
  125         (Just vc, Just uc) ->
  126             let finalC = insertConstrWith union u vc (insertConstrWith union v uc c)
  127             in putConstraints finalC
  128 copyConstraints _ = return ()

  147 -- instantiate a type scheme to a monotype
  148 -- by substituting all bound variables with new free ones
  149 inst :: TypeScheme -> Parser SymbolType
  150 inst (MonoType t)  = return t
  151 inst (AbsType v t) = do
  152     v' <- freshTVar
  153     copyConstraints (TVar v, v')
  154     let substt = substScheme v v' t
  155     inst substt
  156 
  157 -- generalize a monotype to a type scheme
  158 -- by bounding all free variables not found
  159 -- in scope
  160 gen :: S.Set Int -> SymbolType -> TypeScheme
  161 gen freeVars t =
  162     let varNotInScope :: Int -> [Int]
  163         varNotInScope v = ([v | not (S.member v freeVars)])
  164         alg :: TypeF [Int] -> [Int]
  165         alg (FunType f1 f2) = f1 ++ f2
  166         alg (ArrayType _ f) = f
  167         alg (RefType f)     = f
  168         alg _               = []
  169         removeDuplicates :: S.Set Int -> [Int] -> [Int]
  170         removeDuplicates _ []     = []
  171         removeDuplicates s (x:xs) = if S.member x s
  172                                     then removeDuplicates s xs
  173                                     else x:removeDuplicates (S.insert x s) xs
  174         varsNotInScope = cata (alg, varNotInScope) t
  175         varsToBound = removeDuplicates S.empty varsNotInScope
  176     in foldr AbsType (MonoType t) varsToBound

  194 -- Resolution utils
  195 resolveFreeVars :: S.Set Int -> Parser (S.Set Int)
  196 resolveFreeVars fv = do
  197     f <- getUnifier
  198     let g v s = case f (TVar v) of
  199             Just st -> return $ S.union s (S.fromList $ tvarsInType st)
  200             Nothing -> throwSem $ "Unable to resolve type var " ++ pretty (TVar v)
  201     foldrM g S.empty fv

  203 resolveType :: SymbolType -> Parser SymbolType
  204 resolveType st = do
  205     f <- getUnifier
  206     case f st of
  207         Just t  -> return t
  208         Nothing -> throwSem $ "Unable to resolve type " ++ pretty st

  210 resolveNodeType :: Node n => n SemanticTag -> Parser (n SemanticTag)
  211 resolveNodeType n = case typeInfo (tag n) of
  212     NodeType t -> do
  213         rt <- resolveType t
  214         let mn = mapTag (\tg -> tg{typeInfo = NodeType rt}) n
  215         return mn
  216     _          -> do
  217         let p = posn $ tag n
  218         throwSemAtPosn "Unable to compute type of node" p
  219 
  220 resolveTypeScheme :: TypeScheme -> Parser TypeScheme
  221 resolveTypeScheme s = do
  222     f <- getUnifier
  223     aux f s where
  224         aux :: Unifier -> TypeScheme -> Parser TypeScheme
  225         aux f (MonoType t')  = case f t' of
  226             Just t  -> return (MonoType t)
  227             Nothing -> throwSem $ "Unable to resolve type" ++ pretty t'
  228         aux f (AbsType v s') =
  229             let g t@(TVar _)   = if t == TVar v then Just t else f t
  230                 g (SymType tf) = SymType <$> mapM g tf
  231             in AbsType v <$> aux g s'
  232 
  233 resolveTag :: SemanticTag -> Parser SemanticTag
  234 resolveTag tg = case typeInfo tg of
  235     NodeType t -> do
  236         rt <- resolveType t
  237         return tg{typeInfo = NodeType rt}
  238     DefType t  -> do
  239         rt <- resolveTypeScheme t
  240         return tg{typeInfo = DefType rt}
  241     NotTypable -> return tg
  242 
  243 resolveNodeRec :: Node n => n SemanticTag -> Parser (n SemanticTag)
  244 resolveNodeRec = mapM resolveTag

  246 resolveTableEntry :: TableEntry -> Parser TableEntry
  247 resolveTableEntry entry = case entry of
  248     MutableEntry t -> do
  249         rt <- resolveType t
  250         let updated = MutableEntry rt
  251         return updated
  252     ArrayEntry t dim -> do
  253         rt <- resolveType t
  254         let updated = ArrayEntry rt dim
  255         return updated
  256     FunEntry scheme ps -> do
  257         rScheme <- resolveTypeScheme scheme
  258         let updated = FunEntry rScheme ps
  259         return updated
  260     ParamEntry t i -> do
  261         rt <- resolveType t
  262         let updated = ParamEntry rt i
  263         return updated
  264     PatternEntry t -> do
  265         rt <- resolveType t
  266         let updated = PatternEntry rt
  267         return updated
  268     -- constructors can not have var types
  269     ConstrEntry {} -> return entry
  270 
  271 -- Find symbol if exists else throw error
  272 findName :: String -> Parser TableEntry
  273 findName k = let
  274     aux k' = do
  275         ns <- getNames
  276         case query k' ns of
  277             Just entry -> return entry
  278             _ -> throwSem ("Symbol " ++ k' ++ " is not in scope")
  279     in do
  280     entry <- aux k
  281     case entry of
  282         MutableEntry t -> do
  283             rt <- resolveType t
  284             let updated = MutableEntry rt
  285             if rt == t then return entry
  286             else do
  287                 updateName k updated
  288                 return updated
  289         ArrayEntry t dim -> do
  290             rt <- resolveType t
  291             let updated = ArrayEntry rt dim
  292             if rt == t then return entry
  293             else do
  294                 updateName k updated
  295                 return updated
  296         FunEntry scheme ps -> do
  297             rScheme <- resolveTypeScheme scheme
  298             let updated = FunEntry rScheme ps
  299             updateName k updated
  300             return updated
  301         ParamEntry t i -> do
  302             rt <- resolveType t
  303             let updated = ParamEntry rt i
  304             if rt == t then return entry
  305             else do
  306                 updateName k updated
  307                 return updated
  308         PatternEntry t -> do
  309             rt <- resolveType t
  310             let updated = PatternEntry rt
  311             if rt == t then return entry
  312             else do
  313                 updateName k updated
  314                 return updated
  315         -- constructors can not have var types
  316         ConstrEntry {} -> return entry

  318 findType :: Identifier -> Parser [(ConstrIdentifier, [ConstType])]
  319 findType i = do
  320     ts <- getTypes
  321     case query i ts of
  322         Just (TypeEntry constrs) -> return constrs
  323         _ -> throwSem ("Type symbol " ++ i ++ " is not in scope")

  325 -- Check that type is in scope else throw error
  326 checkTypeInScope :: Identifier -> Parser ()
  327 checkTypeInScope = findType >=> const (return ())
